do -- ESP (v7 rewrite: one Entry object per player, same visuals and same Config keys)
	-- Replaces the whole old `do -- ESP ... end` block. Nothing else in the script needs to change:
	-- the ESP tab, config import/export and the aimlock's red target colour all use the same names.
	Config.esp, Config.espNames, Config.espHealth, Config.espDist = false, true, true, true
	Config.tracers, Config.charms = false, false
	Config.espTeamCheck, Config.espTeamColor, Config.espBox, Config.espRainbow = false, false, false, false
	Config.espSkeleton, Config.espHealthBar, Config.espLook, Config.espArrows, Config.espVisibleOnly = false, false, false, false, false
	Config.espTagScale, Config.espTagBg, Config.espBoxScale, Config.espMaxDist = 0.65, 0.35, 1, 5000

	local RAY_INTERVAL = 0.12 -- seconds between line-of-sight checks per player (was every frame)

	local espFolder = create("Folder", { Name = "FlightESP" }, playerGui)
	local overlay = create("ScreenGui", {
		Name = "FlightTracers", ResetOnSpawn = false, IgnoreGuiInset = true, DisplayOrder = -1,
		ZIndexBehavior = Enum.ZIndexBehavior.Sibling,
	}, playerGui)
	local rayParams = RaycastParams.new()
	rayParams.FilterType = Enum.RaycastFilterType.Exclude

	local BONES = {
		R15 = {
			{ "Head", "UpperTorso" }, { "UpperTorso", "LowerTorso" },
			{ "UpperTorso", "LeftUpperArm" }, { "LeftUpperArm", "LeftLowerArm" }, { "LeftLowerArm", "LeftHand" },
			{ "UpperTorso", "RightUpperArm" }, { "RightUpperArm", "RightLowerArm" }, { "RightLowerArm", "RightHand" },
			{ "LowerTorso", "LeftUpperLeg" }, { "LeftUpperLeg", "LeftLowerLeg" }, { "LeftLowerLeg", "LeftFoot" },
			{ "LowerTorso", "RightUpperLeg" }, { "RightUpperLeg", "RightLowerLeg" }, { "RightLowerLeg", "RightFoot" },
		},
		R6 = {
			{ "Head", "Torso" }, { "Torso", "Left Arm" }, { "Torso", "Right Arm" },
			{ "Torso", "Left Leg" }, { "Torso", "Right Leg" },
		},
	}

	-- Positions a thin frame as a line between two screen points.
	local function drawLine(frame, from, to, thickness, color, transparency)
		local delta = to - from
		local middle = (from + to) / 2
		frame.Size = UDim2.fromOffset(delta.Magnitude, thickness)
		frame.Position = UDim2.fromOffset(middle.X, middle.Y)
		frame.Rotation = math.deg(math.atan2(delta.Y, delta.X))
		frame.BackgroundColor3 = color
		frame.BackgroundTransparency = transparency
		frame.Visible = true
	end

	-- Screen rectangle around a character (scaled by `scale`). Returns nil if any corner is behind the camera.
	local function boxBounds(camera, character, scale)
		local cframe, size = character:GetBoundingBox()
		local half = size / 2 * scale
		local minX, minY, maxX, maxY = math.huge, math.huge, -math.huge, -math.huge
		for sx = -1, 1, 2 do
			for sy = -1, 1, 2 do
				for sz = -1, 1, 2 do
					local point = camera:WorldToViewportPoint(cframe:PointToWorldSpace(Vector3.new(half.X * sx, half.Y * sy, half.Z * sz)))
					if point.Z <= 0 then return nil end
					if point.X < minX then minX = point.X end
					if point.X > maxX then maxX = point.X end
					if point.Y < minY then minY = point.Y end
					if point.Y > maxY then maxY = point.Y end
				end
			end
		end
		return minX, minY, maxX, maxY
	end

	----------------------------------------------------------------------------
	-- Entry: everything drawn for one player
	----------------------------------------------------------------------------
	local Entry = {}
	Entry.__index = Entry

	function Entry.new(target, character, head, humanoid)
		local self = setmetatable({
			target = target, character = character, head = head, humanoid = humanoid,
			shown = false, clear = false, nextRay = 0, bones = {}, boneList = nil,
			filter = nil, filterRoot = nil, lastInfo = nil, lastName = nil,
		}, Entry)
		local objects = {}
		self.objects = objects
		local function own(object) table.insert(objects, object) return object end

		self.highlight = own(create("Highlight", {
			Name = "ESPHighlight", Adornee = character, Enabled = false, FillTransparency = 0.82,
			DepthMode = Enum.HighlightDepthMode.AlwaysOnTop,
		}, character))

		-- Tag: built at a fixed 150x48, shrunk by a UIScale (TAG SIZE), anchored at its bottom edge.
		local tag = own(create("BillboardGui", {
			Name = "ESPTag", Adornee = head, AlwaysOnTop = true, Enabled = false, LightInfluence = 0,
			Size = UDim2.fromOffset(150, 48), StudsOffset = Vector3.new(0, 1.7, 0), SizeOffset = Vector2.new(0, 0.5),
			MaxDistance = math.huge,
		}, espFolder))
		local card = create("Frame", {
			AnchorPoint = Vector2.new(0.5, 1), Position = UDim2.fromScale(0.5, 1), Size = UDim2.fromScale(1, 1),
			BackgroundTransparency = 0.35, BorderSizePixel = 0,
		}, tag)
		self.tag, self.card = tag, card
		self.cardScale = create("UIScale", { Scale = Config.espTagScale }, card)
		self.cardStroke = create("UIStroke", { Thickness = 1, Transparency = 0.2 }, card)
		self.topLine = create("Frame", { Size = UDim2.new(1, 0, 0, 2), BorderSizePixel = 0 }, card)
		self.nameLabel = create("TextLabel", {
			Text = target.DisplayName, Font = Fonts.bold, TextSize = 14, BackgroundTransparency = 1,
			Position = UDim2.fromOffset(0, 4), Size = UDim2.new(1, 0, 0, 16),
		}, card)
		self.nameStroke = create("UIStroke", { Thickness = 1.5 }, self.nameLabel)
		self.track = create("Frame", {
			Position = UDim2.fromOffset(10, 24), Size = UDim2.new(1, -20, 0, 8), BackgroundTransparency = 0.2, BorderSizePixel = 0,
		}, card)
		self.trackStroke = create("UIStroke", { Thickness = 1, Transparency = 0.3 }, self.track)
		self.fill = create("Frame", { Size = UDim2.fromScale(1, 1), BorderSizePixel = 0 }, self.track)
		self.info = create("TextLabel", {
			Font = Fonts.regular, TextSize = 10, BackgroundTransparency = 1,
			Position = UDim2.fromOffset(0, 34), Size = UDim2.new(1, 0, 0, 12),
		}, card)
		local diamond = create("Frame", {
			AnchorPoint = Vector2.new(0.5, 0.5), Position = UDim2.new(0.5, 0, 1, 0), Size = UDim2.fromOffset(8, 8),
			Rotation = 45, BackgroundColor3 = GOLD, BorderSizePixel = 0,
		}, card)
		self.diamondStroke = create("UIStroke", { Thickness = 1.5, Transparency = 0.1 }, diamond)

		-- Screen-space pieces (all live in the overlay gui).
		local function pixel(properties)
			properties.BorderSizePixel = 0
			properties.Visible = false
			return own(create("Frame", properties, overlay))
		end
		self.tracer = pixel({ AnchorPoint = Vector2.new(0.5, 0.5) })
		self.box = pixel({ BackgroundTransparency = 1 })
		self.boxStroke = create("UIStroke", { Thickness = 1.5, Transparency = 0.1 }, self.box)
		self.barBack = pixel({ BackgroundTransparency = 0.35 })
		self.barFill = create("Frame", {
			AnchorPoint = Vector2.new(0, 1), Position = UDim2.fromScale(0, 1), Size = UDim2.fromScale(1, 1), BorderSizePixel = 0,
		}, self.barBack)
		self.look = pixel({ AnchorPoint = Vector2.new(0.5, 0.5) })
		self.arrow = pixel({ AnchorPoint = Vector2.new(0.5, 0.5), Rotation = 45, Size = UDim2.fromOffset(14, 14) })
		self.arrowStroke = create("UIStroke", { Color = Color3.new(1, 1, 1), Thickness = 1.5, Transparency = 0.2 }, self.arrow)
		self.arrowLine = pixel({ AnchorPoint = Vector2.new(0.5, 0.5) })
		return self
	end

	function Entry:hide()
		if not self.shown then return end
		self.shown = false
		self.highlight.Enabled, self.tag.Enabled = false, false
		self.tracer.Visible, self.box.Visible, self.barBack.Visible = false, false, false
		self.look.Visible, self.arrow.Visible, self.arrowLine.Visible = false, false, false
		for _, bone in ipairs(self.bones) do bone.Visible = false end
	end

	function Entry:destroy()
		for _, object in ipairs(self.objects) do object:Destroy() end
		for _, bone in ipairs(self.bones) do bone:Destroy() end
		table.clear(self.objects)
		table.clear(self.bones)
	end

	-- Bone parts are looked up once per character, not every frame.
	function Entry:drawSkeleton(camera, color)
		local list = self.boneList
		if not list then
			local rig = BONES[self.character:FindFirstChild("UpperTorso") and "R15" or "R6"]
			list = {}
			for _, pair in ipairs(rig) do
				local a, b = self.character:FindFirstChild(pair[1]), self.character:FindFirstChild(pair[2])
				if a and b and a:IsA("BasePart") and b:IsA("BasePart") then table.insert(list, { a, b }) end
			end
			self.boneList = list
		end
		local used = 0
		for _, pair in ipairs(list) do
			local a, b = pair[1], pair[2]
			if a.Parent and b.Parent then
				local pa = camera:WorldToViewportPoint(a.Position)
				local pb = camera:WorldToViewportPoint(b.Position)
				if pa.Z > 0 and pb.Z > 0 then
					used += 1
					local bone = self.bones[used]
					if not bone then
						bone = create("Frame", { AnchorPoint = Vector2.new(0.5, 0.5), BorderSizePixel = 0 }, overlay)
						self.bones[used] = bone
					end
					drawLine(bone, Vector2.new(pa.X, pa.Y), Vector2.new(pb.X, pb.Y), 1.5, color, 0.15)
				end
			end
		end
		for index = used + 1, #self.bones do self.bones[index].Visible = false end
	end

	function Entry:update(ctx)
		local target, humanoid, head, character, camera = self.target, self.humanoid, self.head, self.character, ctx.camera
		if not (head.Parent and humanoid.Parent) or humanoid.Health <= 0 then return self:hide() end

		local headPos = head.Position
		local distance = ctx.myRoot and (headPos - ctx.myRoot.Position).Magnitude or 0
		if (Config.espTeamCheck and isTeammate(target)) or (Config.espMaxDist < 5000 and distance > Config.espMaxDist) then
			return self:hide()
		end

		-- Line of sight, refreshed a few times a second per player instead of every frame.
		local clear = false
		if Config.charms or Config.tracers or Config.espVisibleOnly then
			if ctx.now >= self.nextRay then
				self.nextRay = ctx.now + RAY_INTERVAL
				if self.filterRoot ~= ctx.myChar then
					self.filterRoot = ctx.myChar
					self.filter = ctx.myChar and { character, ctx.myChar } or { character }
				end
				rayParams.FilterDescendantsInstances = self.filter
				local from = camera.CFrame.Position
				self.clear = workspace:Raycast(from, headPos - from, rayParams) == nil
			end
			clear = self.clear
		end
		if Config.espVisibleOnly and not clear then return self:hide() end
		self.shown = true

		-- Colour: rainbow, then team colour, then the theme accent. Clear sight is gold, the aimlock target is red.
		local custom = Config.espRainbow or (Config.espTeamColor and target.Team ~= nil)
		local base = Colors.accent
		if Config.espRainbow then
			base = ctx.rainbow
		elseif Config.espTeamColor and target.Team then
			base = target.TeamColor.Color
		end
		local locked = Actions.lockedCharacter == character
		local tint = locked and RED or (clear and GOLD or base)

		local highlight = self.highlight
		highlight.Enabled = true
		if Config.charms then
			highlight.FillColor, highlight.FillTransparency, highlight.OutlineColor = tint, 0.25, tint
		else
			highlight.FillColor, highlight.FillTransparency = base, 0.82
			highlight.OutlineColor = locked and RED or (custom and base or Colors.line)
		end
		highlight.OutlineTransparency = 0.05 + 0.5 * (1 - Config.glow)

		local ratio = math.clamp(humanoid.Health / math.max(humanoid.MaxHealth, 1), 0, 1)
		local point, onScreen = camera:WorldToViewportPoint(headPos)

		-- Tracer
		if Config.tracers and onScreen then
			drawLine(self.tracer, ctx.origin, Vector2.new(point.X, point.Y), 2, tint, 0.2 + 0.6 * (1 - Config.glow))
		else
			self.tracer.Visible = false
		end

		-- Box and health bar share one bounding rectangle.
		local minX, minY, maxX, maxY
		if Config.espBox or Config.espHealthBar then
			minX, minY, maxX, maxY = boxBounds(camera, character, Config.espBoxScale)
		end
		if Config.espBox and minX then
			local box = self.box
			box.Position = UDim2.fromOffset(minX, minY)
			box.Size = UDim2.fromOffset(maxX - minX, maxY - minY)
			self.boxStroke.Color = tint
			box.Visible = true
		else
			self.box.Visible = false
		end
		if Config.espHealthBar and minX then
			local back = self.barBack
			back.Position = UDim2.fromOffset(minX - 7, minY)
			back.Size = UDim2.fromOffset(4, maxY - minY)
			back.BackgroundColor3 = Colors.panel
			self.barFill.Size = UDim2.fromScale(1, ratio)
			self.barFill.BackgroundColor3 = Color3.fromHSV(ratio * 0.33, 0.85, 1)
			back.Visible = true
		else
			self.barBack.Visible = false
		end

		-- Skeleton
		if Config.espSkeleton then
			self:drawSkeleton(camera, tint)
		elseif #self.bones > 0 and self.bones[1].Visible then
			for _, bone in ipairs(self.bones) do bone.Visible = false end
		end

		-- Look line: where the head is facing.
		if Config.espLook then
			local from = camera:WorldToViewportPoint(headPos)
			local to = camera:WorldToViewportPoint(headPos + head.CFrame.LookVector * 6)
			if from.Z > 0 and to.Z > 0 then
				drawLine(self.look, Vector2.new(from.X, from.Y), Vector2.new(to.X, to.Y), 2, tint, 0.2)
			else
				self.look.Visible = false
			end
		else
			self.look.Visible = false
		end

		-- Off-screen arrow near the screen edge, pulsing.
		if Config.espArrows and not onScreen then
			local offset = camera.CFrame:PointToObjectSpace(headPos)
			local direction = Vector2.new(offset.X, -offset.Y)
			direction = direction.Magnitude > 0.001 and direction.Unit or Vector2.new(0, 1)
			local center = ctx.screen / 2
			local tip = center + Vector2.new(direction.X * (center.X - 50), direction.Y * (center.Y - 50))
			local pulse = 12 + 4 * math.sin(ctx.now * 6)
			local arrow = self.arrow
			arrow.Position = UDim2.fromOffset(tip.X, tip.Y)
			arrow.Size = UDim2.fromOffset(pulse, pulse)
			arrow.BackgroundColor3, self.arrowStroke.Color = tint, Colors.text
			arrow.Visible = true
			drawLine(self.arrowLine, tip - direction * 22, tip, 2, tint, 0.2)
		else
			self.arrow.Visible, self.arrowLine.Visible = false, false
		end

		-- Tag
		local tag = self.tag
		tag.Enabled = ctx.anyText
		if ctx.anyText then
			self.cardScale.Scale = Config.espTagScale
			local card = self.card
			card.BackgroundColor3, card.BackgroundTransparency = Colors.panel, Config.espTagBg
			self.cardStroke.Color = custom and base or Colors.line
			self.topLine.BackgroundColor3 = base
			self.topLine.BackgroundTransparency = ctx.lineTransparency
			local nameLabel = self.nameLabel
			nameLabel.TextColor3 = Colors.text
			nameLabel.Visible = Config.espNames
			if self.lastName ~= target.DisplayName then
				self.lastName = target.DisplayName
				nameLabel.Text = target.DisplayName
			end
			self.nameStroke.Color, self.nameStroke.Transparency = base, ctx.strokeTransparency
			self.track.BackgroundColor3, self.trackStroke.Color = Colors.panel, Colors.line
			self.track.Visible = Config.espHealth
			self.fill.Size = UDim2.fromScale(ratio, 1)
			self.fill.BackgroundColor3 = ratio < 0.3 and RED or Colors.text
			self.diamondStroke.Color = locked and RED or base

			local text = ""
			if Config.espDist and ctx.myRoot then text = string.format("%d studs", math.floor(distance)) end
			if Config.espHealth then
				local hp = string.format("%d HP", math.floor(humanoid.Health))
				text = text ~= "" and (text .. "  |  " .. hp) or hp
			end
			local info = self.info
			info.Visible = text ~= ""
			info.TextColor3 = Colors.muted
			if self.lastInfo ~= text then
				self.lastInfo = text
				info.Text = text
			end
		end
	end

	----------------------------------------------------------------------------
	-- Tracking players
	----------------------------------------------------------------------------
	local entries = {}

	local function removeEntry(target)
		local entry = entries[target]
		if entry then
			entries[target] = nil
			entry:destroy()
		end
	end

	local function addEntry(target, character)
		local head = character:WaitForChild("Head", 5)
		local humanoid = character:WaitForChild("Humanoid", 5)
		if not (head and humanoid) or target.Character ~= character then return end
		removeEntry(target)
		entries[target] = Entry.new(target, character, head, humanoid)
	end

	local function watch(target)
		if target == player then return end
		target.CharacterAdded:Connect(function(character) task.spawn(addEntry, target, character) end)
		target.CharacterRemoving:Connect(function() removeEntry(target) end)
		if target.Character then task.spawn(addEntry, target, target.Character) end
	end
	for _, other in ipairs(Players:GetPlayers()) do watch(other) end
	Players.PlayerAdded:Connect(watch)
	Players.PlayerRemoving:Connect(removeEntry)

	----------------------------------------------------------------------------
	-- Frame loop
	----------------------------------------------------------------------------
	local ctx = {} -- reused every frame
	local applied = false

	RunService.RenderStepped:Connect(function()
		if not Config.esp then
			if applied then
				applied = false
				for _, entry in pairs(entries) do entry:hide() end
			end
			return
		end
		applied = true

		local camera = workspace.CurrentCamera
		if not camera then return end
		local _, myRoot = getParts()
		local screen = camera.ViewportSize

		ctx.camera, ctx.myChar, ctx.myRoot = camera, player.Character, myRoot
		ctx.now = os.clock()
		ctx.screen, ctx.origin = screen, Vector2.new(screen.X / 2, screen.Y)
		ctx.rainbow = Color3.fromHSV((ctx.now * 0.2) % 1, 0.75, 1)
		ctx.anyText = Config.espNames or Config.espHealth or Config.espDist
		ctx.lineTransparency = 1 - 0.85 * Config.glow
		ctx.strokeTransparency = 1 - 0.5 * Config.glow

		for _, entry in pairs(entries) do entry:update(ctx) end
	end)
end
