--[[
	Flight Status Menu (System Edition)  -  v6 (advanced flight + merged MOVE panel)
	LocalScript. Place in StarterPlayer > StarterPlayerScripts.

	Layout
		Left sidebar tabs: STATUS, MOVE, WORLD, ESP, AIM, LIST, MISC, CONFIG, SETTINGS, EXTRAS, INTERFACE
		Drag the window by its header. Close with X, reopen with the MENU tab or the menu key.
		Tabs with more content than fits scroll with the mouse wheel.

	Controls
		F                  Toggle flight (rebindable)
		RightShift         Show or hide the menu (rebindable)
		W A S D            Move relative to the camera
		Space / Left Ctrl  Ascend / descend (rebindable, MOVE tab)
		Left Shift         Boost while held (also the sprint key)
		Q                  Hold to aimlock, or press to toggle (rebindable, AIM tab; AIMLOCK must be on)
		Z                  Hold to zoom (rebindable, MOVE tab; ZOOM must be on)
		Left Alt + Click   Click teleport (rebindable, MOVE tab; CLICK TP must be on)
		E                  Dash (MOVE tab; DASH must be on)
		N / B / J          Noclip / Fullbright / ESP hotkeys (EXTRAS tab; HOTKEYS must be on)
		Delete             Panic: reset all settings and hide the menu

	v6 additions
		MOVE tab: flight, walking, camera and every movement key now live in one panel.
		ADVANCED FLIGHT: PLANAR, CRUISE, FLY NOCLIP, WHEEL SPEED, BANK TILT, separate BRAKING.
		ANTI OVERRIDE: the flight rig is rebuilt if another script removes it, the humanoid state and
		anchoring are re-asserted every Heartbeat, velocity is written directly as a fallback, and the
		fly key still works when another binding sinks the key. This covers other LocalScripts and
		Roblox state changes. It cannot stop the server from moving or kicking you.

	v4 additions
		AIM tab: target priority, max distance, toggle mode, sticky, skip shield, target marker.
		v5 aimlock: locks a player (survives respawns), runs last, HARD LOCK, OVERRIDE CAM, LOCK GRACE.
		ESP tab: skeleton, HP bar, look line, off-screen arrows, visible only. Aimlock target shows red.
		INTERFACE tab (LIVING UI): edge light, aurora, breathe, corner pulses, animation speed.

	Rank crest (STATUS tab, PLAYER card)
		A rotating magical seal with orbiting glyphs replaces the plain rank letter.
		E / D / C follow the theme, B is emerald, A is amethyst, S is gold with a third seal square.
		Higher tiers show more orbiting glyphs and a stronger glow. GLOW INTENSITY scales it.

	Configs
		The CONFIG tab saves named presets for this session and exports/imports them as a text code.
		Roblox LocalScripts cannot write files, so saved presets are lost when you leave the game.
		Copy the exported code somewhere to keep it.

	A key is required before anything loads (see the Key system section). Owner keys unlock every
	limit, member keys are capped. Note: this check runs on the client, so it can be bypassed by
	anyone who edits their own copy of the script.

	Intro: set INTRO_MUSIC_ID / INTRO_IMAGE below. Click or press any key to skip it.
]]

local function createPalette(hue)
	return {
		accent = Color3.fromHSV(hue, 0.75, 1),
		line = Color3.fromHSV(hue, 0.45, 1),
		text = Color3.fromHSV(hue, 0.10, 1),
		muted = Color3.fromHSV(hue, 0.35, 0.78),
		panel = Color3.fromHSV(hue, 0.85, 0.12),
		backTop = Color3.fromHSV(hue, 0.85, 0.16),
		backBottom = Color3.fromHSV(hue, 0.92, 0.05),
		trim = Color3.fromHSV(hue, 0.30, 1),
	}
end

local Colors = createPalette(Config.hue)
local bindings, glowBindings = {}, {}

local function bind(object, property, role)
	local function apply()
		object[property] = typeof(role) == "function" and role(Colors) or Colors[role]
	end
	apply()
	table.insert(bindings, apply)
	return object
end

local function bindGlow(object, property, transparency)
	local function apply()
		object[property] = 1 - (1 - transparency) * Config.glow
	end
	apply()
	table.insert(glowBindings, apply)
	return object
end

local function setHue(hue)
	Config.hue = hue
	Colors = createPalette(hue)
	for _, apply in ipairs(bindings) do apply() end
	for _, render in pairs(renderers) do render() end
end

local function applyGlow()
	for _, apply in ipairs(glowBindings) do apply() end
end

local Fonts = { regular = Enum.Font.GothamMedium, bold = Enum.Font.GothamBold, black = Enum.Font.GothamBlack }

local function isPress(input)
	return input.UserInputType == Enum.UserInputType.MouseButton1 or input.UserInputType == Enum.UserInputType.Touch
end

local function isMove(input)
	return input.UserInputType == Enum.UserInputType.MouseMovement or input.UserInputType == Enum.UserInputType.Touch
end

local function outline(object, transparency)
	return bind(create("UIStroke", { Thickness = 1, Transparency = transparency }, object), "Color", "line")
end

-- `Glow` (boolean) and `Role` (palette key) are handled here and are not Instance properties.
local function label(parent, properties)
	local glow, role = properties.Glow, properties.Role
	properties.Glow, properties.Role = nil, nil
	properties.BackgroundTransparency = 1
	properties.Font = properties.Font or Fonts.regular
	properties.TextSize = properties.TextSize or 14
	properties.TextXAlignment = properties.TextXAlignment or Enum.TextXAlignment.Left
	properties.Size = properties.Size or UDim2.fromScale(1, 1)
	local object = bind(create("TextLabel", properties, parent), "TextColor3", role or "text")
	if glow then
		bindGlow(bind(create("UIStroke", { Thickness = 1.5 }, object), "Color", "accent"), "Transparency", 0.5)
	end
	return object
end

local function addButton(parent, properties)
	local customFill = properties.BackgroundColor3
	properties.AutoButtonColor = false
	properties.BackgroundTransparency = properties.BackgroundTransparency or 0.3
	properties.BorderSizePixel = 0

	local button = bind(create("TextButton", properties, parent), "TextColor3", "text")
	if not customFill then bind(button, "BackgroundColor3", "panel") end
	outline(button, 0.45)
	return button
end

local function onHold(button, action)
	local token
	button.MouseButton1Down:Connect(function()
		local id = {}
		token = id
		action()
		task.delay(0.4, function()
			while token == id do
				action()
				task.wait(0.06)
			end
		end)
	end)
	local function release() token = nil end
	button.MouseButton1Up:Connect(release)
	button.MouseLeave:Connect(release)
end

-- Drags `target` by `handle` and keeps the window on screen.
local function makeDraggable(handle, target)
	local origin, start
	handle.InputBegan:Connect(function(input)
		if not isPress(input) then return end
		origin, start = input.Position, target.Position
		input.Changed:Connect(function()
			if input.UserInputState == Enum.UserInputState.End then origin = nil end
		end)
	end)
	UserInputService.InputChanged:Connect(function(input)
		if origin and isMove(input) then
			local delta = input.Position - origin
			local view = workspace.CurrentCamera.ViewportSize
			local x = math.clamp(start.X.Offset + delta.X, -view.X / 2 + 80, view.X / 2 - 80)
			local y = math.clamp(start.Y.Offset + delta.Y, -view.Y / 2 + 80, view.Y / 2 - 80)
			target.Position = UDim2.new(start.X.Scale, x, start.Y.Scale, y)
		end
	end)
end

-- Soft light made of stacked translucent diamonds (squares turned 45 degrees, so no curves).
local function addBloom(parent, position, size, layers, transparency, color)
	for index = 1, layers do
		local side = size * index / layers / 1.4142 -- keeps the diamond's tip-to-tip width equal to the old circle diameter
		local circle = create("Frame", {
			AnchorPoint = Vector2.new(0.5, 0.5), Position = position, Rotation = 45,
			Size = UDim2.fromOffset(side, side), BorderSizePixel = 0,
		}, parent)
		if color then
			circle.BackgroundColor3, circle.BackgroundTransparency = color, transparency
		else
			bindGlow(bind(circle, "BackgroundColor3", "accent"), "BackgroundTransparency", transparency)
		end
	end
end

-- Square L-shaped brackets on the four corners. `color` is a palette role or a Color3.
-- A negative inset puts them outside the parent, a positive one inside.
local function addBrackets(parent, length, thickness, color, inset)
	for _, corner in ipairs({ { 0, 0 }, { 1, 0 }, { 0, 1 }, { 1, 1 } }) do
		local cx, cy = corner[1], corner[2]
		for _, horizontal in ipairs({ true, false }) do
			local frame = create("Frame", {
				AnchorPoint = Vector2.new(cx, cy),
				Position = UDim2.new(cx, cx == 0 and inset or -inset, cy, cy == 0 and inset or -inset),
				Size = horizontal and UDim2.fromOffset(length, thickness) or UDim2.fromOffset(thickness, length),
				BorderSizePixel = 0,
			}, parent)
			if typeof(color) == "string" then bind(frame, "BackgroundColor3", color) else frame.BackgroundColor3 = color end
		end
	end
end

-- Rotating rune: three nested squares and a ring. `fixed` (a colour table) ignores the theme.
local sigils = {}
local function makeSigil(parent, size, position, fixed)
	local function paint(object, property, role)
		if fixed then object[property] = fixed[role] else bind(object, property, role) end
		return object
	end
	local function soften(stroke, transparency)
		if fixed then stroke.Transparency = transparency else bindGlow(stroke, "Transparency", transparency) end
	end

	local holder = create("Frame", {
		AnchorPoint = Vector2.new(0.5, 0.5), Position = position,
		Size = UDim2.fromOffset(size, size), BackgroundTransparency = 1,
	}, parent)

	local layers = {}
	for index = 1, 3 do
		local scale = 0.72 - (index - 1) * 0.16
		local frame = create("Frame", {
			AnchorPoint = Vector2.new(0.5, 0.5), Position = UDim2.fromScale(0.5, 0.5),
			Size = UDim2.fromScale(scale, scale), BackgroundTransparency = 1,
		}, holder)
		local stroke = paint(create("UIStroke", { Thickness = index == 2 and 1.5 or 2 }, frame), "Color", index == 2 and "trim" or "accent")
		soften(stroke, 0.15)
		layers[index] = { frame = frame, base = index * 17, speed = (index % 2 == 0 and -1 or 1) * (10 + index * 8) }
	end

	local ring = create("Frame", {
		AnchorPoint = Vector2.new(0.5, 0.5), Position = UDim2.fromScale(0.5, 0.5),
		Size = UDim2.fromScale(0.68, 0.68), Rotation = 45, BackgroundTransparency = 1,
	}, holder)
	soften(paint(create("UIStroke", { Thickness = 1.5 }, ring), "Color", "line"), 0.35)

	local core = paint(create("Frame", {
		AnchorPoint = Vector2.new(0.5, 0.5), Position = UDim2.fromScale(0.5, 0.5),
		Size = UDim2.fromScale(0.08, 0.08), Rotation = 45, BorderSizePixel = 0,
	}, holder), "BackgroundColor3", "text")

	local sigil = { holder = holder }
	function sigil.update(angle)
		for _, layer in ipairs(layers) do layer.frame.Rotation = layer.base + angle * layer.speed end
	end
	return sigil
end

-- Shooting stars with a bright head, tapering tail and four-point flare.
-- `options` sets length, speed, life and interval, each as { min, max }.
local function makeMeteorField(parent, width, height, count, options, fixed)
	local function paint(object, property, role)
		if fixed then object[property] = fixed[role] else bind(object, property, role) end
		return object
	end
	local function fade(object, rotation, points)
		local keypoints = {}
		for _, point in ipairs(points) do table.insert(keypoints, NumberSequenceKeypoint.new(point[1], point[2])) end
		create("UIGradient", { Rotation = rotation, Transparency = NumberSequence.new(keypoints) }, object)
	end

	local meteors = {}
	for index = 1, count do
		local group = create("CanvasGroup", {
			AnchorPoint = Vector2.new(0.5, 0.5), Size = UDim2.fromOffset(200, 40),
			BackgroundTransparency = 1, GroupTransparency = 1, Visible = false,
		}, parent)

		local function strip(thickness, role, points, scaleX, offsetX)
			local frame = paint(create("Frame", {
				AnchorPoint = Vector2.new(1, 0.5), Position = UDim2.new(1, -20, 0.5, 0),
				Size = UDim2.new(scaleX, offsetX, 0, thickness), BorderSizePixel = 0,
			}, group), "BackgroundColor3", role)
			fade(frame, 0, points)
		end
		strip(10, "accent", { { 0, 1 }, { 1, 0.72 } }, 1, -20)
		strip(3, "line", { { 0, 1 }, { 0.6, 0.65 }, { 1, 0.05 } }, 1, -20)
		strip(1.5, "text", { { 0, 1 }, { 1, 0 } }, 0.45, 0)

		for _, spec in ipairs({ { 28, "accent", 0.75 }, { 16, "line", 0.4 }, { 7, "text", 0 } }) do
			local circle = paint(create("Frame", {
				AnchorPoint = Vector2.new(0.5, 0.5), Position = UDim2.new(1, -20, 0.5, 0), Rotation = 45,
				Size = UDim2.fromOffset(spec[1] / 1.4142, spec[1] / 1.4142), BackgroundTransparency = spec[3], BorderSizePixel = 0,
			}, group), "BackgroundColor3", spec[2])
		end
		for _, vertical in ipairs({ false, true }) do
			local flare = paint(create("Frame", {
				AnchorPoint = Vector2.new(0.5, 0.5), Position = UDim2.new(1, -20, 0.5, 0),
				Size = vertical and UDim2.fromOffset(2, 34) or UDim2.fromOffset(34, 2), BorderSizePixel = 0,
			}, group), "BackgroundColor3", "text")
			fade(flare, vertical and 90 or 0, { { 0, 1 }, { 0.5, 0 }, { 1, 1 } })
		end
		meteors[index] = { group = group, active = false }
	end

	local function launch(meteor)
		local mirrored = random:NextInteger(0, 1) == 1
		local angle = random:NextNumber(18, 40)
		if mirrored then angle = 180 - angle end
		local radians = math.rad(angle)
		meteor.direction = Vector2.new(math.cos(radians), math.sin(radians))
		meteor.length = random:NextNumber(options.length[1], options.length[2])
		meteor.speed = random:NextNumber(options.speed[1], options.speed[2])
		meteor.life = random:NextNumber(options.life[1], options.life[2])
		meteor.age = 0
		meteor.head = Vector2.new(
			(mirrored and random:NextNumber(0.4, 1.05) or random:NextNumber(-0.05, 0.6)) * width,
			random:NextNumber(-0.1, 0.35) * height
		)
		meteor.group.Size = UDim2.fromOffset(meteor.length + 20, 40)
		meteor.group.Rotation = angle
		meteor.group.Visible = true
		meteor.active = true
	end

	local field = { enabled = true }
	local timer = random:NextNumber(0.3, 1)

	function field.burst(amount)
		for _, meteor in ipairs(meteors) do
			if amount <= 0 then break end
			if not meteor.active then
				launch(meteor)
				amount -= 1
			end
		end
	end

	function field.update(dt)
		timer -= dt
		if timer <= 0 then
			timer = random:NextNumber(options.interval[1], options.interval[2])
			if field.enabled then field.burst(1) end
		end
		local intensity = fixed and 1 or Config.glow
		for _, meteor in ipairs(meteors) do
			if meteor.active then
				meteor.age += dt
				local progress = meteor.age / meteor.life
				if progress >= 1 then
					meteor.active = false
					meteor.group.Visible = false
				else
					meteor.head += meteor.direction * meteor.speed * dt
					local center = meteor.head - meteor.direction * (meteor.length / 2 - 10)
					meteor.group.Position = UDim2.fromOffset(center.X, center.Y)
					local alpha = math.clamp(math.min(progress / 0.15, (1 - progress) / 0.35), 0, 1)
					meteor.group.GroupTransparency = 1 - alpha * intensity
				end
			end
		end
	end

	return field
end

-- Straight line segments joined end to end (used for the lightning notch). Points are Vector2s in `parent` pixels.
local function addPolyline(parent, points, thickness, role, transparency)
	for index = 1, #points - 1 do
		local from, to = points[index], points[index + 1]
		local delta = to - from
		local middle = (from + to) / 2
		bind(create("Frame", {
			AnchorPoint = Vector2.new(0.5, 0.5), Position = UDim2.fromOffset(middle.X, middle.Y),
			Size = UDim2.fromOffset(delta.Magnitude + thickness, thickness),
			Rotation = math.deg(math.atan2(delta.Y, delta.X)), BackgroundTransparency = transparency, BorderSizePixel = 0,
		}, parent), "BackgroundColor3", role)
	end
end

-- A titled card: translucent panel, accent tick, and a divider under the title with a lightning notch at its right end.
local function addCard(parent, title, x, y, w, h)
	local frame = bind(create("Frame", {
		Position = UDim2.fromOffset(x, y), Size = UDim2.fromOffset(w, h),
		BackgroundTransparency = 0.45, BorderSizePixel = 0, ClipsDescendants = true,
	}, parent), "BackgroundColor3", "panel")
	outline(frame, 0.55)
	bindGlow(bind(create("Frame", { Position = UDim2.fromOffset(0, 5), Size = UDim2.fromOffset(3, 18), BorderSizePixel = 0 }, frame),
		"BackgroundColor3", "accent"), "BackgroundTransparency", 0.05)
	if title then
		label(frame, {
			Text = title, TextSize = 11, Font = Fonts.bold, Role = "trim",
			Position = UDim2.fromOffset(12, 6), Size = UDim2.new(1, -24, 0, 16),
		})
		local notch = w - 10 - 44
		addPolyline(frame, {
			Vector2.new(10, 26), Vector2.new(notch, 26), Vector2.new(notch + 9, 26), Vector2.new(notch + 17, 14),
			Vector2.new(notch + 23, 33), Vector2.new(notch + 31, 20), Vector2.new(notch + 36, 26), Vector2.new(w - 10, 26),
		}, 1.5, "line", 0.45)
	end
	return frame
end

local function addToggle(parent, definition, position, size)
	local button = addButton(parent, { Text = definition.label, TextSize = 12, Position = position, Size = size })
	local function render()
		local enabled = Config[definition.key]
		button.BackgroundColor3 = enabled and Colors.accent or Colors.panel
		button.BackgroundTransparency = enabled and 0.45 or 0.3
		button.TextColor3 = enabled and Colors.text or Colors.muted
	end
	button.Activated:Connect(function()
		local enabled = not Config[definition.key]
		Config[definition.key] = enabled
		if Handlers[definition.key] then Handlers[definition.key](enabled) end
		if definition.key ~= "flying" then
			notify(definition.label .. (enabled and " ENABLED" or " DISABLED"))
		end
		render()
	end)
	renderers[definition.key] = render
	render()
	return button
end

	local track = create("Frame", {
		Position = UDim2.fromOffset(x, y + 22), Size = UDim2.fromOffset(w, 8),
		BackgroundColor3 = Color3.new(1, 1, 1), BorderSizePixel = 0,
	}, parent)
	outline(track, 0.5)

	local fill
	if rainbow then
		local keypoints = {}
		for step = 0, 6 do table.insert(keypoints, ColorSequenceKeypoint.new(step / 6, Color3.fromHSV(step / 6, 0.75, 1))) end
		create("UIGradient", { Color = ColorSequence.new(keypoints) }, track)
	else
		bind(track, "BackgroundColor3", "panel")
		fill = bind(create("Frame", { Size = UDim2.fromScale(0, 1), BorderSizePixel = 0 }, track), "BackgroundColor3", "accent")
	end

	local knob = bind(create("Frame", {
		AnchorPoint = Vector2.new(0.5, 0.5), Size = UDim2.fromOffset(6, 18), BorderSizePixel = 0,
	}, track), "BackgroundColor3", "text")
	outline(knob, 0.3)

	local function render()
		local ratio = math.clamp((get() - min) / (max - min), 0, 1)
		knob.Position = UDim2.fromScale(ratio, 0.5)
		if fill then fill.Size = UDim2.fromScale(ratio, 1) end
		readout.Text = format(get())
	end

	local dragging = false
	local function update(px)
		set(min + (max - min) * math.clamp((px - track.AbsolutePosition.X) / track.AbsoluteSize.X, 0, 1))
		render()
	end
	local hit = create("TextButton", {
		Text = "", BackgroundTransparency = 1, Position = UDim2.fromOffset(x, y + 12), Size = UDim2.fromOffset(w, 28),
	}, parent)
	hit.InputBegan:Connect(function(input)
		if isPress(input) then dragging = true update(input.Position.X) end
	end)
	UserInputService.InputChanged:Connect(function(input)
		if dragging and isMove(input) then update(input.Position.X) end
	end)
	UserInputService.InputEnded:Connect(function(input)
		if isPress(input) then dragging = false end
	end)
	table.insert(refreshers, render)
	render()
end

local function addKeybind(parent, caption, key, x, y, w, h)
	local button = addButton(parent, { TextSize = 12, Position = UDim2.fromOffset(x, y), Size = UDim2.fromOffset(w, h) })
	local function refresh() button.Text = string.format("%s: %s", caption, Config[key].Name) end
	button.Activated:Connect(function()
		button.Text = caption .. ": ..."
		listening = function(newKey)
			if newKey then Config[key] = newKey end
			refresh()
		end
	end)
	table.insert(refreshers, refresh)
	refresh()
	return button
end

local function sha256(message)
	local H = {
		0x6a09e667, 0xbb67ae85, 0x3c6ef372, 0xa54ff53a,
		0x510e527f, 0x9b05688c, 0x1f83d9ab, 0x5be0cd19,
	}
	local length = #message
	local bits = length * 8
	message = message .. "\128" .. string.rep("\0", (55 - length) % 64)
		.. string.char(0, 0, 0, 0, bit32.rshift(bits, 24) % 256, bit32.rshift(bits, 16) % 256, bit32.rshift(bits, 8) % 256, bits % 256)

	for chunk = 1, #message, 64 do
		local w = {}
		for index = 1, 16 do
			local first = chunk + (index - 1) * 4
			local b1, b2, b3, b4 = string.byte(message, first, first + 3)
			w[index] = b1 * 0x1000000 + b2 * 0x10000 + b3 * 0x100 + b4
		end
		for index = 17, 64 do
			local x, y = w[index - 15], w[index - 2]
			local s0 = bit32.bxor(bit32.rrotate(x, 7), bit32.rrotate(x, 18), bit32.rshift(x, 3))
			local s1 = bit32.bxor(bit32.rrotate(y, 17), bit32.rrotate(y, 19), bit32.rshift(y, 10))
			w[index] = (w[index - 16] + s0 + w[index - 7] + s1) % 0x100000000
		end

		local a, b, c, d, e, f, g, h = H[1], H[2], H[3], H[4], H[5], H[6], H[7], H[8]
		for index = 1, 64 do
			local S1 = bit32.bxor(bit32.rrotate(e, 6), bit32.rrotate(e, 11), bit32.rrotate(e, 25))
			local ch = bit32.bxor(bit32.band(e, f), bit32.band(bit32.bnot(e), g))
			local temp1 = (h + S1 + ch + SHA_K[index] + w[index]) % 0x100000000
			local S0 = bit32.bxor(bit32.rrotate(a, 2), bit32.rrotate(a, 13), bit32.rrotate(a, 22))
			local maj = bit32.bxor(bit32.band(a, b), bit32.band(a, c), bit32.band(b, c))
			local temp2 = (S0 + maj) % 0x100000000
			h, g, f, e, d, c, b, a = g, f, e, (d + temp1) % 0x100000000, c, b, a, (temp1 + temp2) % 0x100000000
		end

		local working = { a, b, c, d, e, f, g, h }
		for index = 1, 8 do H[index] = (H[index] + working[index]) % 0x100000000 end
	end

	local parts = {}
	for index = 1, 8 do
		parts[index] = string.format("%04x%04x", bit32.rshift(H[index], 16), bit32.band(H[index], 0xFFFF))
	end
	return table.concat(parts)
end

local function checkKey(text)
	local normalized = (string.upper(text):gsub("%s", ""))
	if #normalized < 8 or #normalized > 64 then return nil, "INVALID KEY" end
	local entry = KEYS[sha256(KEY_SALT .. normalized)]
	if not entry then return nil, "INVALID KEY" end
	if entry.expires and os.time() > entry.expires then return nil, "KEY EXPIRED" end
	return entry
end

local function getParts()
	local character = player.Character
	return character and character:FindFirstChildOfClass("Humanoid"), character and character:FindFirstChild("HumanoidRootPart")
end

-- Keys are ignored while a text box has focus so typing never moves the character.
local function isDown(key)
	return not UserInputService:GetFocusedTextBox() and UserInputService:IsKeyDown(key)
end

local function axis(positive, negative)
	return (isDown(positive) and 1 or 0) - (isDown(negative) and 1 or 0)
end

local function applyHumanoidStats()
	local humanoid = getParts()
	if humanoid then
		humanoid.WalkSpeed = Config.walkSpeed
		humanoid.UseJumpPower = true
		humanoid.JumpPower = Config.jumpPower
	end
end

local function applyStat(key)
	if not ready then return end
	if key == "walkSpeed" or key == "jumpPower" then
		statsModified = true
		applyHumanoidStats()

	end
end

-- All flight internals live in one table so the main chunk gets only one extra local.
local Flight = { name = "FlightControl", rig = nil, yaw = CFrame.identity }

function Flight.destroy()
	for _, object in ipairs(flightObjects) do object:Destroy() end
	flightObjects, Flight.rig = {}, nil
end

function Flight.build(root)
	Flight.destroy()
	local attachment = create("Attachment", { Name = "FlightAttachment" }, root)
	local linear = create("LinearVelocity", {
		Attachment0 = attachment, RelativeTo = Enum.ActuatorRelativeTo.World,
		MaxForce = math.huge, VectorVelocity = velocity,
	}, root)
	local align = create("AlignOrientation", {
		Mode = Enum.OrientationAlignmentMode.OneAttachment, Attachment0 = attachment,
		MaxTorque = math.huge, Responsiveness = 30, CFrame = Flight.yaw,
	}, root)
	flightObjects = { attachment, linear, align }
	Flight.rig = { attachment = attachment, linear = linear, align = align }
end

-- ANTI OVERRIDE: runs every Heartbeat while flying. Repairs anything another local script undid.
function Flight.guard()
	if not Config.flyLocked then return end
	local humanoid, root = getParts()
	if not (humanoid and root) or humanoid.Health <= 0 then return end

	local rig = Flight.rig
	if not rig or rig.attachment.Parent ~= root or rig.linear.Parent ~= root or rig.align.Parent ~= root then
		Flight.build(root)
		rig = Flight.rig
	end
	rig.linear.Enabled, rig.linear.Attachment0, rig.linear.MaxForce = true, rig.attachment, math.huge
	rig.align.Enabled, rig.align.Attachment0, rig.align.MaxTorque = true, rig.attachment, math.huge

	if not humanoid.PlatformStand then humanoid.PlatformStand = true end
	if humanoid:GetState() ~= Enum.HumanoidStateType.PlatformStanding then
		humanoid:ChangeState(Enum.HumanoidStateType.PlatformStanding)
	end
	if root.Anchored and not Config.freeze then root.Anchored = false end
	root.AssemblyLinearVelocity = velocity -- fallback if the LinearVelocity is ever disabled mid-frame
end

-- Input and motion. Bound at the last render priority so it runs after other scripts.
function Flight.step(dt)
	local humanoid, root = getParts()
	local camera = workspace.CurrentCamera
	local rig = Flight.rig
	if not (humanoid and root and camera and rig) or not root.Parent then return end
	local cframe = camera.CFrame

	local look, right = cframe.LookVector, cframe.RightVector
	if Config.flyPlanar then
		look, right = Vector3.new(look.X, 0, look.Z), Vector3.new(right.X, 0, right.Z)
		look = look.Magnitude > 0.001 and look.Unit or Vector3.zero
		right = right.Magnitude > 0.001 and right.Unit or Vector3.zero
	end

	local forward = axis(Enum.KeyCode.W, Enum.KeyCode.S)
	if Config.flyCruise then forward = isDown(Enum.KeyCode.S) and 0 or 1 end -- S brakes while cruising
	local direction = look * forward + right * axis(Enum.KeyCode.D, Enum.KeyCode.A)
	if direction.Magnitude > 0 then direction = direction.Unit end

	local boost = isDown(Enum.KeyCode.LeftShift) and Config.boost or 1
	local vertical = axis(Config.upKey, Config.downKey)
	local target = (direction * Config.flySpeed + Vector3.yAxis * vertical * Config.verticalSpeed) * boost

	local rate = 1
	if Config.smooth then
		rate = target.Magnitude < velocity.Magnitude and Config.flyDecel or Config.flyAccel
		rate = 1 - (1 - rate) ^ (dt * 60) -- same feel at any frame rate
	end
	velocity = velocity:Lerp(target, rate)
	rig.linear.VectorVelocity = velocity

	local flat = Vector3.new(cframe.LookVector.X, 0, cframe.LookVector.Z)
	if Config.faceCamera and flat.Magnitude > 0.001 then Flight.yaw = CFrame.lookAt(Vector3.zero, flat) end

	local orientation = Flight.yaw
	local tilt = math.rad(Config.flyTilt)
	if tilt > 0 then
		local localVelocity = Flight.yaw:VectorToObjectSpace(velocity)
		local lean = math.clamp(-localVelocity.Z / math.max(Config.flySpeed, 1), -1, 1)
		local bank = math.clamp(localVelocity.X / math.max(Config.flySpeed, 1), -1, 1)
		orientation = Flight.yaw * CFrame.Angles(-lean * tilt, 0, -bank * tilt)
	end
	rig.align.CFrame = orientation
end

local function stopFlight()
	if flightConnection then flightConnection:Disconnect() flightConnection = nil end
	pcall(function() RunService:UnbindFromRenderStep(Flight.name) end)
	Flight.destroy()
	velocity = Vector3.zero
	local humanoid = getParts()
	if humanoid then humanoid.PlatformStand = false end
	-- Give collisions back if FLY NOCLIP was the only thing holding them off.
	if not Config.noclip and Handlers.noclip then Handlers.noclip(false) end
end

local function startFlight()
	local humanoid, root = getParts()
	if not (humanoid and root) then return end
	humanoid.PlatformStand = true

	local flat = Vector3.new(root.CFrame.LookVector.X, 0, root.CFrame.LookVector.Z)
	Flight.yaw = flat.Magnitude > 0.001 and CFrame.lookAt(Vector3.zero, flat) or CFrame.identity
	Flight.build(root)

	flightConnection = RunService.Heartbeat:Connect(Flight.guard) -- also the "is flying" flag
	pcall(function() RunService:UnbindFromRenderStep(Flight.name) end)
	RunService:BindToRenderStep(Flight.name, Enum.RenderPriority.Last.Value, Flight.step)
end

local function setFlying(enabled)
	local before = Config.flying
	if enabled then
		if not flightConnection then startFlight() end
	else
		stopFlight()
	end
	Config.flying = flightConnection ~= nil
	if renderers.flying then renderers.flying() end
	if Config.flying ~= before then
		notify(Config.flying and "FLIGHT ACTIVATED" or "FLIGHT DEACTIVATED")
	end
end

local function setNoclip(enabled)
	if enabled then return end
	for part in pairs(noclipParts) do
		if part.Parent then part.CanCollide = true end
	end
	table.clear(noclipParts)
end

Handlers.flying, Handlers.noclip = setFlying, setNoclip

-- Turning FLY NOCLIP off mid-flight must restore collisions.
Handlers.flyNoclip = function(enabled)
	if not enabled and not Config.noclip then setNoclip(false) end
end

			noclipParts[part] = true
			part.CanCollide = false
		end
	end
end)

		saved[instance] = saved[instance] or {}
		if saved[instance][property] == nil then saved[instance][property] = instance[property] end
		instance[property] = value
	end
	local function restore(instance, property)
		local store = saved[instance]
		if store and store[property] ~= nil then
			instance[property] = store[property]
			store[property] = nil
		end
	end

	Config.freezeTime, Config.noEffects, Config.noFx = false, false, false
	Config.shadows = Lighting.GlobalShadows
	Config.clockTime = Lighting.ClockTime
	Config.gravity = workspace.Gravity
	Config.exposure = Lighting.ExposureCompensation
	Config.quality = 5
	pcall(function()
		local level = settings().Rendering.QualityLevel.Value
		Config.quality = math.clamp(level == 0 and 5 or level, 1, 10)
	end)

	Handlers.fullbright = function(enabled)
		if enabled then
			change(Lighting, "Brightness", 2)
			change(Lighting, "Ambient", Color3.fromRGB(178, 178, 178))
			change(Lighting, "OutdoorAmbient", Color3.fromRGB(178, 178, 178))
		else
			restore(Lighting, "Brightness")
			restore(Lighting, "Ambient")
			restore(Lighting, "OutdoorAmbient")
		end
	end
	Handlers.shadows = function(enabled) Lighting.GlobalShadows = enabled end
	Handlers.noFog = function(enabled)
		local atmosphere = Lighting:FindFirstChildOfClass("Atmosphere")
		if enabled then
			change(Lighting, "FogEnd", 1e6)
			if atmosphere then change(atmosphere, "Density", 0) change(atmosphere, "Haze", 0) end
		else
			restore(Lighting, "FogEnd")
			if atmosphere then restore(atmosphere, "Density") restore(atmosphere, "Haze") end
		end
	end
	Handlers.lite = function(enabled)
		Config.particles, Config.stars = not enabled, not enabled
		if renderers.particles then renderers.particles() end
		if renderers.stars then renderers.stars() end
	end

	-- Numeric settings: the sliders and the config loader both go through these.
	Handlers.quality = function(value)
		pcall(function() settings().Rendering.QualityLevel = Enum.QualityLevel:FromValue(value) end)
	end
	Handlers.clockTime = function(value) Lighting.ClockTime = value end
	Handlers.gravity = function(value) workspace.Gravity = value end
	Handlers.exposure = function(value) Lighting.ExposureCompensation = value end

	-- NO EFFECTS: turns off blur, bloom, colour correction, sun rays and depth of field.
	local effects = {}
	Handlers.noEffects = function(enabled)
		if enabled then
			for _, child in ipairs(Lighting:GetChildren()) do
				if child:IsA("PostEffect") then
					effects[child] = child.Enabled
					child.Enabled = false
				end
			end
		else
			for child, was in pairs(effects) do
				if child.Parent then child.Enabled = was end
			end
			table.clear(effects)
		end
	end

	-- NO PARTICLES: switches off every particle emitter, trail and beam in the world.
	local fxSaved, fxConnection = {}, nil
	local function silence(instance)
		if (instance:IsA("ParticleEmitter") or instance:IsA("Trail") or instance:IsA("Beam")) and fxSaved[instance] == nil then
			fxSaved[instance] = instance.Enabled
			instance.Enabled = false
		end
	end
	Handlers.noFx = function(enabled)
		if enabled then
			for _, instance in ipairs(workspace:GetDescendants()) do silence(instance) end
			if not fxConnection then fxConnection = workspace.DescendantAdded:Connect(silence) end
		else
			if fxConnection then fxConnection:Disconnect() fxConnection = nil end
			for instance, was in pairs(fxSaved) do
				if instance.Parent then instance.Enabled = was end
			end
			table.clear(fxSaved)
		end
	end

	-- FREEZE TIME keeps the clock on the TIME OF DAY slider.
	RunService.Heartbeat:Connect(function()
		if Config.freezeTime then Lighting.ClockTime = Config.clockTime end
	end)
end

	-- Screen-space rectangle around a character, shrunk or grown by `scale`. Nil when any corner is behind the camera.
	local function boxBounds(camera, character, scale)
		local cframe, size = character:GetBoundingBox()
		local half = size / 2 * scale
		local minX, minY, maxX, maxY = math.huge, math.huge, -math.huge, -math.huge
		for _, sx in ipairs({ -1, 1 }) do
			for _, sy in ipairs({ -1, 1 }) do
				for _, sz in ipairs({ -1, 1 }) do
					local point = camera:WorldToViewportPoint(cframe:PointToWorldSpace(Vector3.new(half.X * sx, half.Y * sy, half.Z * sz)))
					if point.Z <= 0 then return nil end
					minX, maxX = math.min(minX, point.X), math.max(maxX, point.X)
					minY, maxY = math.min(minY, point.Y), math.max(maxY, point.Y)
				end
			end
		end
		return minX, minY, maxX, maxY
	end

	-- Draws the bone lines for one character (R15 or R6), reusing a pool of frames.
	local function drawSkeleton(entry, camera, color)
		local character = entry.character
		local list = character:FindFirstChild("UpperTorso") and R15_BONES or R6_BONES
		local used = 0
		for _, pair in ipairs(list) do
			local a, b = character:FindFirstChild(pair[1]), character:FindFirstChild(pair[2])
			if a and b and a:IsA("BasePart") and b:IsA("BasePart") then
				local pa = camera:WorldToViewportPoint(a.Position)
				local pb = camera:WorldToViewportPoint(b.Position)
				if pa.Z > 0 and pb.Z > 0 then
					used += 1
					local bone = entry.bones[used]
					if not bone then
						bone = create("Frame", { AnchorPoint = Vector2.new(0.5, 0.5), BorderSizePixel = 0 }, tracerGui)
						entry.bones[used] = bone
					end
					drawLine(bone, Vector2.new(pa.X, pa.Y), Vector2.new(pb.X, pb.Y), 1.5, color, 0.15)
				end
			end
		end
		for index = used + 1, #entry.bones do entry.bones[index].Visible = false end
	end

				entry.barBack.Visible = true
			else
				entry.barBack.Visible = false
			end

			if Config.espSkeleton then
				drawSkeleton(entry, camera, tint)
			else
				for _, bone in ipairs(entry.bones) do bone.Visible = false end
			end

				if from.Z > 0 and to.Z > 0 then
					drawLine(entry.look, Vector2.new(from.X, from.Y), Vector2.new(to.X, to.Y), 2, tint, 0.2)
				else
					entry.look.Visible = false
				end
			else
				entry.look.Visible = false
			end

			-- Off-screen arrow on an ellipse near the screen edge, pulsing.
			if Config.espArrows and not onScreen then
				local offset = camera.CFrame:PointToObjectSpace(entry.head.Position)
				local direction = Vector2.new(offset.X, -offset.Y)
				direction = direction.Magnitude > 0.001 and direction.Unit or Vector2.new(0, 1)
				local center = screenSize / 2
				local tip = center + Vector2.new(direction.X * (center.X - 50), direction.Y * (center.Y - 50))
				local pulse = 12 + 4 * math.sin(os.clock() * 6)
				entry.arrow.Position = UDim2.fromOffset(tip.X, tip.Y)
				entry.arrow.Size = UDim2.fromOffset(pulse, pulse)
				entry.arrow.BackgroundColor3, entry.arrowStroke.Color = tint, Colors.text
				entry.arrow.Visible = true
				drawLine(entry.arrowLine, tip - direction * 22, tip, 2, tint, 0.2)
			else
				entry.arrow.Visible, entry.arrowLine.Visible = false, false
			end

			entry.tag.Enabled = anyText
			if anyText then
				entry.cardScale.Scale = Config.espTagScale
				entry.card.BackgroundColor3 = Colors.panel
				entry.card.BackgroundTransparency = Config.espTagBg
				entry.cardStroke.Color = custom and base or Colors.line
				entry.topLine.BackgroundColor3 = base
				entry.topLine.BackgroundTransparency = lineTransparency
				entry.nameLabel.TextColor3 = Colors.text
				entry.nameLabel.Visible = Config.espNames

				entry.trackStroke.Color = Colors.line
				entry.track.Visible = Config.espHealth
				entry.fill.Size = UDim2.fromScale(ratio, 1)
				entry.fill.BackgroundColor3 = ratio < 0.3 and RED or Colors.text
				entry.diamondStroke.Color = locked and RED or base

				local parts = {}
				if Config.espDist and myRoot then
					table.insert(parts, string.format("%d studs", math.floor(distance)))
				end
				if Config.espHealth then
					table.insert(parts, string.format("%d HP", math.floor(humanoid.Health)))
				end
				entry.info.Visible = #parts > 0
				entry.info.TextColor3 = Colors.muted
				entry.info.Text = table.concat(parts, "  |  ")
			end
		end
	end)

	-- Returns the aim part, the character and the humanoid, or nil when the player is not a valid target.
	local function validPart(other)

		if not Config.aimWalls then return true end
		local filter = { character }
		if player.Character then table.insert(filter, player.Character) end
		wallParams.FilterDescendantsInstances = filter
		local from = camera.CFrame.Position
		return workspace:Raycast(from, part.Position - from, wallParams) == nil
	end

	-- The aim part of `other` if they can be locked right now, otherwise nil.
	local function usable(other, camera)
		if not targetAllowed(other) then return nil end
		local part, character = validPart(other)
		if not part then return nil end
		if (part.Position - camera.CFrame.Position).Magnitude > Config.aimMaxDist then return nil end
		if not visible(part, character, camera) then return nil end
		return part
	end

	local function aimCenter(camera)
		return UserInputService.MouseBehavior == Enum.MouseBehavior.LockCenter
			and camera.ViewportSize / 2 or UserInputService:GetMouseLocation()
	end

	-- Lowest score wins: screen distance, world distance or current health, depending on TARGET PRIORITY.
	local function pickTarget(camera)
		local center = aimCenter(camera)
		local best, bestScore = nil, math.huge
		for _, other in ipairs(Players:GetPlayers()) do
			if targetAllowed(other) then
				local part, character, humanoid = validPart(other)
				if part then
					local worldDistance = (part.Position - camera.CFrame.Position).Magnitude
					local point, onScreen = camera:WorldToViewportPoint(part.Position)
					if onScreen and worldDistance <= Config.aimMaxDist then
						local screenDistance = (Vector2.new(point.X, point.Y) - center).Magnitude
						if screenDistance <= Config.aimRadius and visible(part, character, camera) then
							local score = screenDistance
							if Config.aimPriority == 2 then
								score = worldDistance
							elseif Config.aimPriority == 3 then
								score = humanoid.Health
							end
							if score < bestScore then best, bestScore = other, score end
						end
					end
				end
			end
		end
		return best
	end

	-- OVERRIDE CAM: while locked the camera is set to Scriptable so the default camera script stops writing to it.
	-- A Scriptable camera does not follow the character, so the offset from the character is kept by hand.
	local function takeCamera(camera, rootPart)
		if takenCamera ~= camera then
			takenCamera, savedType = camera, camera.CameraType
			followOffset = (rootPart and savedType ~= Enum.CameraType.Scriptable)
				and camera.CFrame.Position - rootPart.Position or nil
		end
		if camera.CameraType ~= Enum.CameraType.Scriptable then
			camera.CameraType = Enum.CameraType.Scriptable
		end
	end

	local function releaseCamera()
		local camera = takenCamera
		takenCamera, followOffset = nil, nil
		if camera and camera.Parent and savedType and savedType ~= Enum.CameraType.Scriptable
			and camera.CameraType == Enum.CameraType.Scriptable then
			camera.CameraType = savedType
		end
	end

	-- TOGGLE MODE: the aim key flips the lock on and off.
	UserInputService.InputBegan:Connect(function(input, processed)
		if processed or listening or introPlaying or not Config.aimToggleMode then return end
		if input.KeyCode == Config.aimKey then toggled = not toggled end
	end)

	RunService:BindToRenderStep("FlightAimlock", Enum.RenderPriority.Last.Value, function(dt)
		local camera = workspace.CurrentCamera

		local showRing = Config.aimFovCircle and Config.aimlock and camera ~= nil and not introPlaying
		fovRing.Visible = showRing
		if showRing then
			local center = aimCenter(camera)
			fovRing.Position = UDim2.fromOffset(center.X, center.Y)
			local side = Config.aimRadius * 2 / 1.4142 -- diamond whose tips sit on the lock radius
			fovRing.Size = UDim2.fromOffset(side, side)
		end

		if not Config.aimToggleMode then toggled = false end
		local pressed
		if Config.aimToggleMode then
			pressed = toggled
		else
			pressed = UserInputService:IsKeyDown(Config.aimKey)
		end
		local humanoid, myRoot = getParts()
		local active = Config.aimlock and camera and humanoid and humanoid.Health > 0 and not introPlaying
			and not UserInputService:GetFocusedTextBox() and pressed
		if not active then
			lockedPlayer, lostAt, Actions.lockedCharacter = nil, nil, nil
			marker.Visible = false
			releaseCamera()
			return
		end

		if takenCamera and takenCamera ~= camera then releaseCamera() end

		-- Keep the same player across respawns; drop them only if they left or are no longer allowed.
		if lockedPlayer and (lockedPlayer.Parent ~= Players or not targetAllowed(lockedPlayer)) then
			lockedPlayer, lostAt = nil, nil
		end
		if not Config.aimSticky then lockedPlayer, lostAt = nil, nil end

		local part = lockedPlayer and usable(lockedPlayer, camera)
		if part then
			lostAt = nil
		elseif lockedPlayer then
			-- Dead, hidden or between respawns: hold on for the grace time, then pick someone new.
			lostAt = lostAt or os.clock()
			if os.clock() - lostAt > Config.aimGrace then lockedPlayer, lostAt = nil, nil end
		end
		if not lockedPlayer then
			lockedPlayer = pickTarget(camera)
			part = lockedPlayer and usable(lockedPlayer, camera)
		end
		Actions.lockedCharacter = lockedPlayer and lockedPlayer.Character or nil

		if not lockedPlayer then
			marker.Visible = false
			releaseCamera()
			return
		end

		-- Camera control: take it (so nothing else can move it) or just write last.
		local current = camera.CFrame
		if Config.aimOverride then
			takeCamera(camera, myRoot)
			if followOffset and myRoot then current = CFrame.new(myRoot.Position + followOffset) * current.Rotation end
		else
			releaseCamera()
		end

		if not part then
			-- Grace period: keep the view where it is.
			marker.Visible = false
			camera.CFrame = current
			return
		end

		-- Spinning marker on the target.
		local screenPoint, onScreen = camera:WorldToViewportPoint(part.Position)
		marker.Visible = Config.aimMarker and onScreen
		marker.Position = UDim2.fromOffset(screenPoint.X, screenPoint.Y)
		marker.Rotation = (os.clock() * 120) % 360

		if os.clock() - pingClock > 0.5 then
			pingClock = os.clock()
			local ok, seconds = pcall(function() return player:GetNetworkPing() end)
			if ok and typeof(seconds) == "number" then pingSeconds = seconds end
		end

		local aimPoint = part.Position
		if Config.aimPing then
			aimPoint += part.AssemblyLinearVelocity * pingSeconds * Config.aimPingScale
		end

		local goal = CFrame.lookAt(current.Position, aimPoint)
		if Config.aimHard or Config.aimSmooth <= 0 then
			camera.CFrame = goal -- HARD LOCK: exact aim every frame
		else
			camera.CFrame = current:Lerp(goal, 1 - Config.aimSmooth ^ (dt * 60))
		end
	end)
end

do -- Movement, camera and utility
	local speedCap = role.caps and role.caps.walkSpeed or math.huge
	local sprintBase, lastSafe
	local savedMaxZoom = player.CameraMaxZoomDistance
	local afkConnection
	local waypoints = {}
	Actions.viewing = nil
	Actions.waypoints = waypoints

	-- Handlers for toggles that change something outside this script.

		local _, rootPart = getParts()

	end

		player.CameraMaxZoomDistance = enabled and 100000 or savedMaxZoom
	end

		player.CameraMode = enabled and Enum.CameraMode.LockFirstPerson or Enum.CameraMode.Classic
	end
	Handlers.zoom = function(enabled)
		local camera = workspace.CurrentCamera
		if camera and not enabled then camera.FieldOfView = Config.fov end
	end
	Handlers.antiAfk = function(enabled)
		if enabled and not afkConnection then
			afkConnection = player.Idled:Connect(function()
				pcall(function()
					local virtualUser = game:GetService("VirtualUser")
					virtualUser:CaptureController()
					virtualUser:ClickButton2(Vector2.zero)
				end)
			end)
		elseif not enabled and afkConnection then
			afkConnection:Disconnect()
			afkConnection = nil
		end
	end

	-- Per-frame movement features.
	RunService.Heartbeat:Connect(function(dt)
		local humanoid, rootPart = getParts()
		if not (humanoid and rootPart) then return end

		if Config.keepStats then
			if not sprintBase then humanoid.WalkSpeed = Config.walkSpeed end
			humanoid.UseJumpPower = true
			humanoid.JumpPower = Config.jumpPower
		end

		-- SPRINT: hold Left Shift to multiply the current walk speed (members stay under their walk cap).
		if Config.sprint and not Config.flying and isDown(Enum.KeyCode.LeftShift) then
			sprintBase = sprintBase or humanoid.WalkSpeed
			humanoid.WalkSpeed = math.max(sprintBase, math.min(sprintBase * Config.sprintMult, speedCap))
		elseif sprintBase then
			humanoid.WalkSpeed = sprintBase
			sprintBase = nil
		end

		if Config.bhop and not Config.flying and isDown(Enum.KeyCode.Space)
			and humanoid.FloorMaterial ~= Enum.Material.Air then
			humanoid.Jump = true
		end

		if Config.glide and not Config.flying then
			local current = rootPart.AssemblyLinearVelocity
			if current.Y < -Config.glideSpeed then
				rootPart.AssemblyLinearVelocity = Vector3.new(current.X, -Config.glideSpeed, current.Z)
			end
		end

		if Config.antiVoid then
			local floorY = workspace.FallenPartsDestroyHeight + 60
			if humanoid.FloorMaterial ~= Enum.Material.Air and rootPart.Position.Y > floorY then
				lastSafe = rootPart.CFrame
			elseif lastSafe and rootPart.Position.Y < floorY then
				rootPart.CFrame = lastSafe + Vector3.new(0, 4, 0)
				rootPart.AssemblyLinearVelocity = Vector3.zero
				notify("ANTI VOID SAVE")
			end
		end

		if Config.spin and not Config.flying then
			rootPart.CFrame *= CFrame.Angles(0, math.rad(Config.spinSpeed * dt), 0)
		end
	end)

	-- Zoom: hold the zoom key to narrow the field of view.
	RunService.RenderStepped:Connect(function(dt)
		local camera = workspace.CurrentCamera
		if camera and Config.zoom then
			local goal = isDown(Config.zoomKey) and Config.zoomFov or Config.fov
			camera.FieldOfView += (goal - camera.FieldOfView) * math.min(1, dt * 12)
		end
	end)

	-- Click teleport: hold the click-TP key and click the world.
	UserInputService.InputBegan:Connect(function(input, processed)
		if processed or introPlaying or not Config.clickTp then return end
		if input.UserInputType ~= Enum.UserInputType.MouseButton1 or not isDown(Config.clickTpKey) then return end
		local camera = workspace.CurrentCamera
		local _, rootPart = getParts()
		if not (camera and rootPart) then return end
		local location = UserInputService:GetMouseLocation()
		local ray = camera:ViewportPointToRay(location.X, location.Y)
		local params = RaycastParams.new()
		params.FilterType = Enum.RaycastFilterType.Exclude
		params.FilterDescendantsInstances = { player.Character }
		local result = workspace:Raycast(ray.Origin, ray.Direction * 5000, params)
		if result then
			rootPart.CFrame = CFrame.new(result.Position + Vector3.new(0, 3, 0)) * rootPart.CFrame.Rotation
			rootPart.AssemblyLinearVelocity = Vector3.zero
		end
	end)

	-- Actions used by buttons in the LIST and MISC tabs.
	function Actions.teleportTo(other)
		local _, rootPart = getParts()
		local target = other.Character and other.Character:FindFirstChild("HumanoidRootPart")
		if rootPart and target then
			rootPart.CFrame = target.CFrame * CFrame.new(0, 2, 4)
			rootPart.AssemblyLinearVelocity = Vector3.zero
			notify("TELEPORTED TO " .. string.upper(other.DisplayName))
		else
			notify("TARGET UNAVAILABLE")
		end
	end

	function Actions.stopView()
		Actions.viewing = nil
		local humanoid = getParts()
		local camera = workspace.CurrentCamera
		if camera and humanoid then camera.CameraSubject = humanoid end
		renderAll()
	end

	function Actions.view(other)
		if Actions.viewing == other then
			Actions.stopView()
			notify("SPECTATE OFF")
			return
		end
		local humanoid = other.Character and other.Character:FindFirstChildOfClass("Humanoid")
		local camera = workspace.CurrentCamera
		if humanoid and camera then
			Actions.viewing = other
			camera.CameraSubject = humanoid
			renderAll()
			notify("VIEWING " .. string.upper(other.DisplayName))
		else
			notify("TARGET UNAVAILABLE")
		end
	end

	function Actions.setWaypoint(index)
		local _, rootPart = getParts()
		if rootPart then
			waypoints[index] = rootPart.CFrame
			renderAll()
			notify("WAYPOINT " .. index .. " SAVED")
		end
	end

	function Actions.gotoWaypoint(index)
		local _, rootPart = getParts()
		if rootPart and waypoints[index] then
			rootPart.CFrame = waypoints[index]
			rootPart.AssemblyLinearVelocity = Vector3.zero
			notify("TELEPORTED TO WAYPOINT " .. index)
		else
			notify("WAYPOINT " .. index .. " IS EMPTY")
		end
	end

	function Actions.resetCharacter()
		local humanoid = getParts()
		if humanoid then humanoid.Health = 0 end
	end

	function Actions.rejoin()
		notify("REJOINING")
		pcall(function() TeleportService:Teleport(game.PlaceId, player) end)
	end

		gradeSat = { -1, 2 }, gradeContrast = { -0.5, 1 }, gradeBright = { -0.5, 0.5 }, blurSize = { 0, 40 },
		uiOpacity = { 0, 0.7 }, notifyTime = { 0.5, 8 }, rainbowSpeed = { 0.02, 1 }, fontFamily = { 1, 5 },
		animSpeed = { 0.2, 3 },
		flyDecel = { 0.02, 1 }, flyTilt = { 0, 45 },
	}) do LIMITS[key] = range end

	-- Colour grade and blur live in Lighting, so NO EFFECTS also switches them off.
	local grade = create("ColorCorrectionEffect", { Name = "FlightGrade" }, Lighting)
	local blur = create("BlurEffect", { Name = "FlightBlur", Size = 0 }, Lighting)
	Handlers.gradeSat = function(value) grade.Saturation = value end
	Handlers.gradeContrast = function(value) grade.Contrast = value end
	Handlers.gradeBright = function(value) grade.Brightness = value end
	Handlers.blurSize = function(value) blur.Size = value end
	Handlers.maxZoom = function(value) player.CameraMaxZoomDistance = value end
	Handlers.dynamicFov = function(enabled)
		local camera = workspace.CurrentCamera
		if camera and not enabled and not Config.zoom then camera.FieldOfView = Config.fov end
	end

	local hipBase = setmetatable({}, { __mode = "k" })
	local trail = {}
	local camTouched, extraJumps = false, 0
	local lastDash, lastWarn, lastGround, recallClock = 0, 0, 0, 0

	RunService.Heartbeat:Connect(function(dt)
		if Config.timeFlow > 0 and not Config.freezeTime then
			Lighting.ClockTime = (Lighting.ClockTime + dt * Config.timeFlow / 60) % 24
		end

		local humanoid, rootPart = getParts()
		if not (humanoid and rootPart) then return end
		local airborne = humanoid.FloorMaterial == Enum.Material.Air
		if not airborne then
			extraJumps = 0
			lastGround = os.clock()
		end

		if Config.hipOffset ~= 0 or hipBase[humanoid] then
			hipBase[humanoid] = hipBase[humanoid] or humanoid.HipHeight
			humanoid.HipHeight = hipBase[humanoid] + Config.hipOffset
		end
		if Config.camOffsetY ~= 0 or camTouched then
			camTouched = Config.camOffsetY ~= 0
			humanoid.CameraOffset = Vector3.new(0, Config.camOffsetY, 0)
		end

		if Config.fastFall and not Config.flying and airborne and isDown(Config.downKey) then
			local current = rootPart.AssemblyLinearVelocity
			if current.Y > -Config.fallSpeed then
				rootPart.AssemblyLinearVelocity = Vector3.new(current.X, -Config.fallSpeed, current.Z)
			end
		end

		if Config.autoWalk and not Config.flying and not UserInputService:GetFocusedTextBox()
			and axis(Enum.KeyCode.W, Enum.KeyCode.S) == 0 and axis(Enum.KeyCode.D, Enum.KeyCode.A) == 0 then
			humanoid:Move(Vector3.new(0, 0, -1), true)
		end

		if Config.lowHpWarn and humanoid.Health > 0 and humanoid.Health / math.max(humanoid.MaxHealth, 1) < 0.25
			and os.clock() - lastWarn > 6 then
			lastWarn = os.clock()
			notify("LOW HEALTH")
		end

		-- Position history for RECALL (one sample every half second, ten samples).
		recallClock += dt
		if recallClock >= 0.5 then
			recallClock = 0
			if not Config.freeze then
				table.insert(trail, rootPart.CFrame)
				if #trail > 10 then table.remove(trail, 1) end
			end
		end
	end)

	-- Speed-based field of view (yields to ZOOM when both are on).
	RunService.RenderStepped:Connect(function(dt)
		if not Config.dynamicFov or Config.zoom then return end
		local camera = workspace.CurrentCamera
		local _, rootPart = getParts()
		if not (camera and rootPart) then return end
		local top = math.max(Config.flySpeed * Config.boost, 1)
		local ratio = math.clamp(rootPart.AssemblyLinearVelocity.Magnitude / top, 0, 1)
		local goal = Config.fov + Config.fovKick * ratio
		camera.FieldOfView += (goal - camera.FieldOfView) * math.min(1, dt * 8)
	end)

	UserInputService.JumpRequest:Connect(function()
		if not Config.doubleJump or Config.flying or extraJumps >= 1 or os.clock() - lastGround < 0.2 then return end
		local humanoid = getParts()
		if humanoid and humanoid.FloorMaterial == Enum.Material.Air then
			extraJumps += 1
			humanoid:ChangeState(Enum.HumanoidStateType.Jumping)
		end
	end)

		if processed or introPlaying or not Config.dash or Config.flying then return end
		if input.KeyCode ~= Config.dashKey or os.clock() - lastDash < Config.dashCooldown then return end
		local humanoid, rootPart = getParts()
		local camera = workspace.CurrentCamera
		if not (humanoid and rootPart and camera) then return end
		local direction = humanoid.MoveDirection
		if direction.Magnitude < 0.1 then
			direction = Vector3.new(camera.CFrame.LookVector.X, 0, camera.CFrame.LookVector.Z)
		end
		if direction.Magnitude < 0.01 then return end
		lastDash = os.clock()
		local current = rootPart.AssemblyLinearVelocity
		rootPart.AssemblyLinearVelocity = direction.Unit * Config.dashPower + Vector3.new(0, math.max(current.Y, 0), 0)
	end)

	-- RESPAWN HERE: come back where you died.

	local function watchCharacter(character)
		table.clear(trail)
		local humanoid = character:WaitForChild("Humanoid", 5)
		local rootPart = character:WaitForChild("HumanoidRootPart", 5)
		if not (humanoid and rootPart) then return end
		local saved = deathCFrame
		deathCFrame = nil
		if Config.respawnInPlace and saved then rootPart.CFrame = saved end
		humanoid.Died:Connect(function() deathCFrame = rootPart.CFrame end)
	end
	player.CharacterAdded:Connect(watchCharacter)
	if player.Character then task.spawn(watchCharacter, player.Character) end

		if Config.joinAlerts then notify(string.upper(other.DisplayName) .. " JOINED") end
	end)
	Players.PlayerRemoving:Connect(function(other)
		if Config.joinAlerts and other ~= player then notify(string.upper(other.DisplayName) .. " LEFT") end
	end)

	function Actions.recall()
		local _, rootPart = getParts()
		local target = trail[1]
		if rootPart and target then
			rootPart.CFrame = target
			rootPart.AssemblyLinearVelocity = Vector3.zero
			table.clear(trail)
			notify("RECALLED")
		else
			notify("NOTHING TO RECALL")
		end
	end

	function Actions.toSpawn()
		local _, rootPart = getParts()
		local spawn = player.RespawnLocation or workspace:FindFirstChildWhichIsA("SpawnLocation", true)
		if rootPart and spawn then
			rootPart.CFrame = spawn.CFrame + Vector3.new(0, 5, 0)
			rootPart.AssemblyLinearVelocity = Vector3.zero
			notify("TELEPORTED TO SPAWN")
		else
			notify("NO SPAWN FOUND")
		end
	end

-- Runtime state that is never saved into a config.
local SKIP = { flying = true, freeze = true, whitelist = true }
local STAT_BY_KEY = {}
for _, stat in ipairs(STATS) do STAT_BY_KEY[stat.key] = stat end

-- Copy of every saveable setting. With `onlyChanged`, only values that differ from the defaults.
local DEFAULT_CONFIG = {}
local function capture(onlyChanged)
	local out = {}
	for key, value in pairs(Config) do
		local kind = typeof(value)
		if not SKIP[key] and (kind == "number" or kind == "boolean" or kind == "string" or kind == "EnumItem") then
			if not onlyChanged or DEFAULT_CONFIG[key] ~= value then out[key] = value end
		end
	end
	return out
end
DEFAULT_CONFIG = capture(false)

-- Turns untrusted data (an imported code) into values that are safe to apply.
local function decodeConfig(raw)
	local clean = {}
	for key, value in pairs(raw) do
		local current = Config[key]
		if SKIP[key] or current == nil then
			-- Unknown or runtime-only setting: ignored.
		elseif typeof(current) == "boolean" then
			if typeof(value) == "boolean" then clean[key] = value end
		elseif typeof(current) == "number" then
			if typeof(value) == "number" then
				local limit, stat = LIMITS[key], STAT_BY_KEY[key]
				if limit then
					clean[key] = math.clamp(value, limit[1], limit[2])
				elseif stat then
					clean[key] = math.clamp(value, stat.min, stat.max)
				end
			end
		elseif typeof(current) == "EnumItem" then
			if typeof(value) == "string" and string.sub(value, 1, 4) == "key:" then
				local ok, keyCode = pcall(function() return Enum.KeyCode[string.sub(value, 5)] end)
				if ok and keyCode then clean[key] = keyCode end
			end
		elseif key == "aimPart" then
			if value == "Head" or value == "HumanoidRootPart" then clean[key] = value end
		end
	end
	return clean
end

local function exportConfig()
	local data = {}
	for key, value in pairs(capture(true)) do
		data[key] = typeof(value) == "EnumItem" and ("key:" .. value.Name) or value
	end
	return HttpService:JSONEncode(data)
end

local function updateParticles(dt, elapsed)
	particleLayer.Visible = Config.particles
	if not Config.particles then return end
	for _, p in ipairs(particles) do
		p.y -= p.speed * Config.flow * dt
		p.x += math.sin(elapsed * p.freq + p.phase) * p.drift * dt
		if p.y < -20 then p.x, p.y = random:NextNumber() * WIN_W, WIN_H + 20 end
		local fade = math.clamp(math.min(p.y, WIN_H - p.y) / 60, 0, 1)
		local twinkle = 0.75 + 0.25 * math.sin(elapsed * p.freq * 3 + p.phase)
		p.frame.Position = UDim2.fromOffset(p.x, p.y)
		p.frame.BackgroundTransparency = 1 - p.alpha * fade * twinkle
	end
	for _, rune in ipairs(Living.runes) do
		rune.y -= rune.speed * Config.flow * dt
		if rune.y < -80 then rune.x, rune.y = random:NextNumber() * WIN_W, WIN_H + 80 end
		rune.frame.Position = UDim2.fromOffset(rune.x + math.sin(elapsed * 0.5 + rune.phase) * 20, rune.y)
		rune.frame.Rotation = elapsed * rune.spin + rune.phase * 57
		local fade = math.clamp(math.min(rune.y, WIN_H - rune.y) / 120, 0, 1)
		rune.stroke.Transparency = 1 - 0.4 * fade * Config.glow
	end
end

-- Edge light, corner pulses and the per-frame update for all the animation above.
local updateLiving
do
	local EDGE_LEN = 220
	local PERIMETER = 2 * (WIN_W + WIN_H)
	local EDGES = {
		{ start = 0, length = WIN_W },                 -- top, left to right
		{ start = WIN_W, length = WIN_H },             -- right, top to bottom
		{ start = WIN_W + WIN_H, length = WIN_W },     -- bottom, right to left
		{ start = 2 * WIN_W + WIN_H, length = WIN_H }, -- left, bottom to top
	}
	local edgeLayers = {}
	for comet = 1, 2 do
		edgeLayers[comet] = {}
		for edgeIndex = 1, 4 do
			local layers = {}
			for layerIndex, spec in ipairs({ { 6, "accent", 0.5 }, { 2, "text", 1 } }) do
				local frame = create("Frame", { BorderSizePixel = 0, Visible = false, ZIndex = 20 }, main)
				bind(frame, "BackgroundColor3", spec[2])
				local gradient = create("UIGradient", { Rotation = ({ 0, 90, 180, 270 })[edgeIndex] }, frame)
				layers[layerIndex] = { frame = frame, gradient = gradient, thickness = spec[1], strength = spec[3] }
			end
			edgeLayers[comet][edgeIndex] = layers
		end
	end

	local function placeEdge(edgeIndex, frame, thickness, a, b)
		local start, length = EDGES[edgeIndex].start, b - a
		if edgeIndex == 1 then
			frame.Position, frame.Size = UDim2.fromOffset(a - start, 0), UDim2.fromOffset(length, thickness)
		elseif edgeIndex == 2 then
			frame.Position, frame.Size = UDim2.fromOffset(WIN_W - thickness, a - start), UDim2.fromOffset(thickness, length)
		elseif edgeIndex == 3 then
			frame.Position, frame.Size = UDim2.fromOffset(WIN_W - (b - start), WIN_H - thickness), UDim2.fromOffset(length, thickness)
		else
			frame.Position, frame.Size = UDim2.fromOffset(0, WIN_H - (b - start)), UDim2.fromOffset(thickness, length)
		end
	end

	local function updateEdges(t)
		local head0 = (t * 260) % PERIMETER
		for comet = 1, 2 do
			local head = (head0 + (comet - 1) * PERIMETER / 2) % PERIMETER
			for edgeIndex, edge in ipairs(EDGES) do
				local layers = edgeLayers[comet][edgeIndex]
				local shown = false
				if Config.edgeChase then
					for _, shift in ipairs({ 0, PERIMETER }) do -- the second pass catches a tail that wrapped past the start
						local h = head + shift
						local a = math.max(h - EDGE_LEN, edge.start)
						local b = math.min(h, edge.start + edge.length)
						if b - a > 0.5 then
							local brightA = ((a - (h - EDGE_LEN)) / EDGE_LEN) ^ 2
							local brightB = ((b - (h - EDGE_LEN)) / EDGE_LEN) ^ 2
							for _, layer in ipairs(layers) do
								placeEdge(edgeIndex, layer.frame, layer.thickness, a, b)
								local k = Config.glow * layer.strength
								layer.gradient.Transparency = NumberSequence.new(1 - brightA * k, 1 - brightB * k)
								layer.frame.Visible = true
							end
							shown = true
							break
						end
					end
				end
				if not shown then
					for _, layer in ipairs(layers) do layer.frame.Visible = false end
				end
			end
		end
	end

	-- Pulsing diamonds on the four outer corners of the window.
	local corners = {}
	for index, corner in ipairs({ { 0, 0 }, { 1, 0 }, { 0, 1 }, { 1, 1 } }) do
		local diamond = create("Frame", {
			AnchorPoint = Vector2.new(0.5, 0.5), Position = UDim2.fromScale(corner[1], corner[2]), Rotation = 45,
			Size = UDim2.fromOffset(8, 8), BorderSizePixel = 0,
		}, root)
		bind(diamond, "BackgroundColor3", "trim")
		corners[index] = diamond
	end

	-- Called every frame from the live readout loop with the menu's elapsed time.
	updateLiving = function(elapsed)
		local t = elapsed * Config.animSpeed

		updateEdges(t)

		Living.aurora.Visible = Config.aurora
		if Config.aurora then
			Living.aurora.GroupTransparency = 1 - 0.85 * Config.glow
			for _, band in ipairs(Living.bands) do
				band.frame.Position = UDim2.fromOffset(
					WIN_W / 2 + math.sin(t * band.speed + band.phase) * 110,
					band.baseY + math.sin(t * band.speed * 1.7 + band.phase) * 50
				)
			end
			Living.scan.Position = UDim2.fromOffset(0, ((t * 0.12) % 1) * (WIN_H + 112) - 56)
		end

		local breath = Config.breathe and (1 + 0.45 * math.sin(t * 1.3)) or 1
		for index, layer in ipairs(Living.halo) do
			layer.Size = UDim2.new(1, index * 7 * breath, 1, index * 7 * breath)
		end

		for index, diamond in ipairs(corners) do
			diamond.Visible = Config.cornerPulse
			if Config.cornerPulse then
				local pulse = 0.5 + 0.5 * math.sin(t * 2.4 + index * 1.57)
				local side = 6 + 6 * pulse
				diamond.Size = UDim2.fromOffset(side, side)
				diamond.BackgroundTransparency = 1 - (0.35 + 0.65 * pulse) * Config.glow
			end
		end
	end
end

local pages, tabNames, currentTab = {}, {}, nil
local headings = {}

local function showTab(name)
	currentTab = name
	for pageName, page in pairs(pages) do page.Visible = pageName == name end
	heading.Text = headings[name]
	for _, tabName in ipairs(tabNames) do
		if renderers["tab" .. tabName] then renderers["tab" .. tabName]() end
	end
end

local function addTab(index, name, title)
	local button = create("TextButton", {
		Text = "", AutoButtonColor = false, BorderSizePixel = 0,
		Position = UDim2.fromOffset(10, 8 + (index - 1) * 30), Size = UDim2.fromOffset(SIDE_W - 20, 27),
	}, sidebar)
	local bar = create("Frame", { Size = UDim2.new(0, 3, 1, 0), BorderSizePixel = 0 }, button)
	local text = label(button, {
		Text = string.format("%02d   %s", index, name), TextSize = 12, Font = Fonts.bold,
		Position = UDim2.fromOffset(14, 0), Size = UDim2.new(1, -14, 1, 0),
	})
	-- Pages scroll when their cards are taller than the window; fitPage sets the canvas once a tab is built.
	local page = create("ScrollingFrame", {
		Position = UDim2.fromOffset(CONTENT_X, CONTENT_Y), Size = UDim2.fromOffset(CONTENT_W + 8, CONTENT_H),
		BackgroundTransparency = 1, BorderSizePixel = 0, Visible = false,
		ScrollBarThickness = 4, CanvasSize = UDim2.new(), ScrollingDirection = Enum.ScrollingDirection.Y,
	}, main)
	bind(page, "ScrollBarImageColor3", "accent")

	local function render()
		local active = currentTab == name
		button.BackgroundColor3 = Colors.accent
		button.BackgroundTransparency = active and 0.72 or 1
		bar.BackgroundColor3 = Colors.accent
		bar.BackgroundTransparency = active and 0 or 1
		text.TextColor3 = active and Colors.text or Colors.muted
	end
	button.Activated:Connect(function() showTab(name) end)
	button.MouseEnter:Connect(function()
		if currentTab ~= name then button.BackgroundTransparency = 0.88 end
	end)
	button.MouseLeave:Connect(render)

	renderers["tab" .. name] = render
	pages[name], headings[name] = page, title
	table.insert(tabNames, name)
	render()
	return page
end

local function fitPage(page)
	local bottom = 0
	for _, child in ipairs(page:GetChildren()) do
		if child:IsA("GuiObject") then
			bottom = math.max(bottom, child.Position.Y.Offset + child.Size.Y.Offset)
		end
	end
	page.CanvasSize = bottom <= CONTENT_H and UDim2.new() or UDim2.fromOffset(0, bottom + 6)
end

-- Widgets that depend on flight code ------------------------------------------

local function addStepper(parent, stat, x, y)
	local row = create("Frame", { Position = UDim2.fromOffset(x, y), Size = UDim2.fromOffset(270, 34), BackgroundTransparency = 1 }, parent)
	label(row, { Text = stat.label, TextSize = 11, Role = "muted", Size = UDim2.fromOffset(100, 32) })
	local value = label(row, {
		TextSize = 22, Font = Fonts.bold, Glow = true, TextXAlignment = Enum.TextXAlignment.Center,
		Position = UDim2.fromOffset(100, 0), Size = UDim2.fromOffset(70, 32),
	})
	local track = bind(create("Frame", {
		Position = UDim2.fromOffset(0, 32), Size = UDim2.fromOffset(270, 2), BackgroundTransparency = 0.5, BorderSizePixel = 0,
	}, row), "BackgroundColor3", "panel")
	local fill = bind(create("Frame", { Size = UDim2.fromScale(0, 1), BorderSizePixel = 0 }, track), "BackgroundColor3", "accent")

	local function set(amount)
		Config[stat.key] = math.clamp(amount, stat.min, stat.max)
		value.Text = tostring(Config[stat.key])
		fill.Size = UDim2.fromScale((Config[stat.key] - stat.min) / (stat.max - stat.min), 1)
		applyStat(stat.key)
	end
	for index, delta in ipairs({ -stat.step, stat.step }) do
		local button = addButton(row, {
			Text = delta < 0 and "-" or "+", TextSize = 18,
			Position = UDim2.fromOffset(184 + (index - 1) * 38, 3), Size = UDim2.fromOffset(32, 26),
		})
		onHold(button, function() set(Config[stat.key] + delta) end)
	end
	steppers[stat.key] = set
	set(Config[stat.key])
end

	if Handlers[key] then Handlers[key](value) end
end

-- Applies a config table (defaults first, then `data`) to the script and refreshes every widget.
local function applyConfig(data)
	local merged = {}
	for key, value in pairs(DEFAULT_CONFIG) do merged[key] = value end
	for key, value in pairs(data) do merged[key] = value end

	-- Only touch settings that actually change (skips the slow font, particle and theme repaints),
	-- and isolate each handler so one failure cannot abort the whole load.
	for key, value in pairs(merged) do
		if steppers[key] then
			steppers[key](value)
		elseif Config[key] ~= value then
			if key == "hue" then
				setHue(value)
			else
				Config[key] = value
				if Handlers[key] then
					local ok, problem = pcall(Handlers[key], value)
					if not ok then warn("[FlightMenu] config handler '" .. key .. "' failed: " .. tostring(problem)) end
				end
			end
		end
	end
	-- Handlers can touch other settings (LITE MODE changes particles and stars); restore the saved values.
	for key, value in pairs(merged) do
		if not steppers[key] then Config[key] = value end
	end

	applyGlow()
	for _, refresh in ipairs(refreshers) do refresh() end
	renderAll()
end

	local nameBox = addTextBox(saveCard, {

	})

	local listCard = addCard(page, "SAVED CONFIGS", 0, 86, CONTENT_W, 214)
	local scroll = create("ScrollingFrame", {
		Position = UDim2.fromOffset(14, 36), Size = UDim2.fromOffset(width, 166),
		BackgroundTransparency = 1, BorderSizePixel = 0, ScrollBarThickness = 4,
		CanvasSize = UDim2.new(), AutomaticCanvasSize = Enum.AutomaticSize.Y,
	}, listCard)
	bind(scroll, "ScrollBarImageColor3", "accent")
	create("UIListLayout", { Padding = UDim.new(0, 4), SortOrder = Enum.SortOrder.LayoutOrder }, scroll)

	-- Built-in presets are stored as differences from the defaults; yours are full copies of the settings at save time.
	local configs = {
		{ name = "DEFAULT", data = DEFAULT_CONFIG, builtin = true },
		{
			name = "ESP PRESET", builtin = true,
			data = { esp = true, espBox = true, espNames = true, espHealth = true, espDist = true, espTeamCheck = true, espTagScale = 0.6 },
		},
		{
			name = "SPEED RUN", builtin = true,
			data = { flySpeed = 250, verticalSpeed = 150, boost = 4, smooth = false, sprint = true, sprintMult = 2, bhop = true, glide = true },
		},
	}
	local MAX_USER_CONFIGS = 25

	-- Persistence: uses writefile/readfile when the environment provides them (survives rejoining),
	-- otherwise configs last for this session only. Everything is pcall-guarded.
	local FILE = "FlightMenuConfigs.json"
	local env = getfenv()
	local canFile = typeof(env.writefile) == "function" and typeof(env.readfile) == "function" and typeof(env.isfile) == "function"
	local canCopy = typeof(env.setclipboard) == "function"

	local function pack(data)
		local out = {}
		for key, value in pairs(data) do
			out[key] = typeof(value) == "EnumItem" and ("key:" .. value.Name) or value
		end
		return out
	end

	local function persist()
		if not canFile then return end
		local list = {}
		for _, entry in ipairs(configs) do
			if not entry.builtin then table.insert(list, { name = entry.name, data = pack(entry.data) }) end
		end
		pcall(function() env.writefile(FILE, HttpService:JSONEncode({ version = 1, configs = list })) end)
	end

	-- Loaded files go through decodeConfig, so a hand-edited file cannot inject bad values.
	local function restore()
		if not canFile then return end
		pcall(function()
			if not env.isfile(FILE) then return end
			local decoded = HttpService:JSONDecode(env.readfile(FILE))
			if typeof(decoded) ~= "table" or typeof(decoded.configs) ~= "table" then return end
			for _, item in ipairs(decoded.configs) do
				if typeof(item) == "table" and typeof(item.name) == "string" and typeof(item.data) == "table"
					and #configs - 3 < MAX_USER_CONFIGS then
					table.insert(configs, { name = string.sub(string.upper(item.name), 1, 24), data = decodeConfig(item.data) })
				end
			end
		end)
	end
	local savedNote = canFile and "" or "  (SESSION ONLY)"

	local function rebuild()
		for _, child in ipairs(scroll:GetChildren()) do
			if child:IsA("Frame") then child:Destroy() end
		end
		for index, entry in ipairs(configs) do
			local row = create("Frame", { LayoutOrder = index, Size = UDim2.new(1, -8, 0, 28), BackgroundTransparency = 1 }, scroll)
			local name = addButton(row, { Text = (entry.builtin and "[ PRESET ]  " or "") .. entry.name, TextSize = 12, Size = UDim2.new(1, entry.builtin and -72 or -140, 1, 0) })
			name.TextXAlignment = Enum.TextXAlignment.Left
			create("UIPadding", { PaddingLeft = UDim.new(0, 10) }, name)
			local load = addButton(row, { Text = "LOAD", TextSize = 11, Position = UDim2.new(1, -68, 0, 0), Size = UDim2.fromOffset(68, 28) })
			load.Activated:Connect(function()
				applyConfig(entry.data)
				notify("CONFIG LOADED: " .. entry.name)
			end)
			name.Activated:Connect(function()
				if not entry.builtin then nameBox.Text = entry.name end
			end)
			if not entry.builtin then
				local delete = addButton(row, { Text = "DELETE", TextSize = 11, Position = UDim2.new(1, -140, 0, 0), Size = UDim2.fromOffset(68, 28) })
				delete.Activated:Connect(function()
					table.remove(configs, index)
					persist()
					rebuild()
					notify("CONFIG DELETED")
				end)
			end
		end
	end

	saveButton.Activated:Connect(function()
		local name = string.upper((string.gsub(nameBox.Text, "^%s*(.-)%s*$", "%1")))
		name = string.sub(name, 1, 24)
		if name == "" then name = "CONFIG " .. (#configs - 2) end
		for _, entry in ipairs(configs) do
			if entry.builtin and entry.name == name then
				notify("THAT NAME IS A PRESET")
				return
			end
		end
		local data = capture(false)
		for _, entry in ipairs(configs) do
			if entry.name == name then
				entry.data = data
				persist()
				notify("CONFIG UPDATED: " .. name .. savedNote)
				return
			end
		end
		if #configs - 3 >= MAX_USER_CONFIGS then
			notify("CONFIG LIST IS FULL")
			return
		end
		table.insert(configs, { name = name, data = data })
		persist()
		nameBox.Text = ""
		rebuild()
		notify("CONFIG SAVED: " .. name .. savedNote)
	end)
	restore()
	rebuild()

	local share = addCard(page, "IMPORT / EXPORT", 0, 310, CONTENT_W, 118)
	local codeBox = addTextBox(share, {
		PlaceholderText = "PASTE A CONFIG CODE HERE, OR PRESS EXPORT",
		Position = UDim2.fromOffset(14, 36), Size = UDim2.fromOffset(width, 30),
	})
	local exportButton = addButton(share, { Text = "EXPORT CURRENT", TextSize = 12, Position = UDim2.fromOffset(14, 76), Size = UDim2.fromOffset(170, 28) })
	local importButton = addButton(share, { Text = "IMPORT CODE", TextSize = 12, Position = UDim2.fromOffset(192, 76), Size = UDim2.fromOffset(170, 28) })
	label(share, {
		Text = canFile and "CONFIGS AUTO-SAVE TO FlightMenuConfigs.json" or "SAVED CONFIGS LAST FOR THIS SESSION ONLY - COPY THE CODE TO KEEP IT", TextSize = 9, Role = "muted",
		TextXAlignment = Enum.TextXAlignment.Right, Position = UDim2.fromOffset(370, 82), Size = UDim2.fromOffset(202, 16),
	})
	exportButton.Activated:Connect(function()
		codeBox.Text = exportConfig()
		if canCopy then pcall(env.setclipboard, codeBox.Text) end
		notify(canCopy and "CODE COPIED" or "CONFIG EXPORTED")
	end)
	importButton.Activated:Connect(function()
		local ok, decoded = pcall(function() return HttpService:JSONDecode(codeBox.Text) end)
		if not ok or typeof(decoded) ~= "table" then
			notify("INVALID CONFIG CODE")
			return
		end
		applyConfig(decodeConfig(decoded))
		notify("CONFIG IMPORTED")
	end)
	fitPage(page)

local function tween(object, duration, properties, style, direction)
	local info = TweenInfo.new(duration, style or Enum.EasingStyle.Quad, direction or Enum.EasingDirection.Out)

		pause(0.8)

		tween(titleScale, 0.5, { Scale = 1 }, Enum.EasingStyle.Back)
		pause(0.4)
		tween(beamGroup, 1.2, { GroupTransparency = 1 })
		pause(1.5)
	end

	sequence()

	rootScale.Scale = targetScale() * 0.9
	setMenuVisible(true)
	tween(rootScale, 0.9, { Scale = targetScale() }, Enum.EasingStyle.Back)
	tween(backdrop, 0.9, { BackgroundTransparency = 1 })
	for _, group in ipairs(groups) do tween(group, 0.8, { GroupTransparency = 1 }) end

	skipConnection:Disconnect()
	introGui:Destroy()

local function rankFor(speed)
	for _, entry in ipairs(RANKS) do
		if speed >= entry[1] then return entry[2] end
	end
	return "E"
end

	ui.hpFill.Size = UDim2.fromScale(math.clamp(health / maxHealth, 0, 1), 1)

	ui.velFill.Size = UDim2.fromScale(math.clamp(speed / math.max(topSpeed, 1), 0, 1), 1)

	ui.z.Text = string.format("%d", math.floor(position.Z))
	ui.top.Text = string.format("%d", topSpeed)

	fps += (1 / math.max(dt, 0.001) - fps) * 0.1
	ui.fps.Text = string.format("%d", math.floor(fps + 0.5))
	if elapsed - pingClock > 0.5 then
		pingClock = elapsed
		local ok, seconds = pcall(function() return player:GetNetworkPing() end)
		if ok and typeof(seconds) == "number" then ping = seconds * 1000 end
	end
	ui.ping.Text = string.format("%dms", math.floor(ping + 0.5))
	local seconds = math.floor(os.clock() - startClock)
	ui.time.Text = string.format("%02d:%02d", math.floor(seconds / 60) % 100, seconds % 60)

	if elapsed - hintClock > 0.5 then
		hintClock = elapsed
		hintLabel.Text = string.format(
			"WASD MOVE   |   %s / %s  UP / DOWN   |   SHIFT BOOST   |   %s FLY   |   %s MENU",
			Config.upKey.Name, Config.downKey.Name, Config.flyKey.Name, Config.menuKey.Name
		)
	end
end)

UserInputService.InputBegan:Connect(function(input, processed)
	if listening then
		if input.UserInputType == Enum.UserInputType.Keyboard then
			local callback = listening
			listening = nil
			callback(input.KeyCode ~= Enum.KeyCode.Escape and input.KeyCode or nil)
		end
		return
	end
	if introPlaying then return end

	if input.KeyCode == Config.flyKey then
		-- With ANTI OVERRIDE on, a game binding that "sinks" the key can't block the fly toggle.
		if (not processed or Config.flyLocked) and not UserInputService:GetFocusedTextBox() then
			setFlying(not Config.flying)
		end
	elseif not processed and input.KeyCode == Config.menuKey then
		setMenuVisible(not root.Visible)
	end
end)

	end

	UserInputService.InputBegan:Connect(function(input, processed)
		if processed or introPlaying or listening then return end
		local code = input.KeyCode
		if code == Config.panicKey then
			applyConfig({})
			setFlying(false)
			setMenuVisible(false)
		elseif Config.hotkeys then
			if code == Config.noclipKey then
				flip("noclip", "NOCLIP")
			elseif code == Config.fullbrightKey then
				flip("fullbright", "FULLBRIGHT")
			elseif code == Config.espKey then
				flip("esp", "ESP")
			end
		end
	end)
end

ready = true

if INTRO_ENABLED then task.spawn(playIntro) else releaseKeyScreen() end
